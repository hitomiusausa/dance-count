## 段3からの揺れの原因（Task 1。「seed 直後の S4 待ちで 30 秒切れ」）

**再現**（Task 1 Step 1 の `seed-race.mjs`。dev 5191・12 並行 × 24 回）: 今までの手順（seed → `page.goto(<S4 のハッシュ>)` → `page.reload()`）は **24 回中 20 回 S4 が開かず S3（`#/routines`）**。直した手順は **0／24（2 回）**。ui:check を 3 本並行（`DC_ONLY=S4,S5`）でも、直す前は 3 本中 2 本が S4 待ちで止まり、直した後は 3 本とも 87 PASS。

**原因（計測で確かめた）**: 保存領域の開きの遅れ（1.5 秒の打ち切り）ではない（失敗した回も開くのは 44〜157ms・ルーティンは DB に 1 件入っている）。**検査の手順の競り合い**: seed はアプリの裏で IndexedDB に書く。続く `page.goto(<同じ文書のハッシュ>)` は**読み込み直しではなくハッシュの変化**なので、動いているアプリはメモリの古い一覧（seed の前＝空）で `resolveScreen` を走らせ、知らないルーティンの S4／S5 を **`replaceRoute` で `#/routines` に置き換える**。その直後の `page.reload()` は置き換わった URL を読み直すので、重い時ほど（アプリのハッシュの処理が reload より先に走るほど）S3 が出る。アプリの作り（知らない ID は S3 へ）は正しく、直すのは検査の手順。

**直し**: `tools/ui-check-routines.js` に `openAfterSeed(page, url)`（`about:blank` を挟んで目当ての URL を新しく読み込む）を足し、seed の後に画面を開く 7 か所（`check-app-ui.js` 1・`smoke-routines.js` 3・`smoke-share.js` 2＝S7・S8 は置き換えないが同じ形にそろえる）を置き換える。`smoke:practice` の「ui:check の直後に 1 回だけ G1 が落ちる」（段4の台帳）は seed を使わないので**別の揺れ**（再現できず。直さない）。

+// 形が読めない（@fontsource の版上げで CSS が変わった）時は投げる＝黙って空の一覧にしない。
+export function parseFontsourceCss(css) {
+  const faces = [];
+  for (const match of css.matchAll(/\/\*\s*([\w[\]-]+?)-(\d{3})-normal\s*\*\/\s*@font-face\s*{([^}]*)}/g)) {
+    const range = match[3].match(/unicode-range:\s*([^;]+);/);
+    if (!range) throw new Error(`no unicode-range for ${match[1]}`);
+    const ranges = range[1].split(',').map(part => {
+      const [from, to] = part.trim().replace(/^U\+/i, '').split('-');
+      const a = Number.parseInt(from, 16);
+      const b = Number.parseInt(to ?? from, 16);
+      if (!Number.isFinite(a) || !Number.isFinite(b)) throw new Error(`bad unicode-range "${part}" in ${match[1]}`);
+      return [a, b];
+    });
+    faces.push({ name: match[1].replace(/[[\]]/g, ''), ranges });
+  }
+  if (faces.length === 0) throw new Error('no @font-face with a unicode-range was found (has the @fontsource CSS changed?)');
+  return faces;
+}

+  const codes = new Set([...text].map(char => char.codePointAt(0)));
+  return faces.filter(face => [...codes].some(code => face.ranges.some(([a, b]) => code >= a && code <= b))).map(face => face.name);
+}

+  const match = file.match(/^assets\/([\w-]+?)-(\d{3})-normal-[\w-]+\.woff2$/);

+}

+  return `assets/{${subsets.join(',')}}-{${weights.join(',')}}-normal-*.woff2`;

+export function readPrecacheList(swText) {
+  const start = swText.indexOf('[{"revision"');
+  if (start < 0) return null;
+  let depth = 0;
+  let inString = false;
+  for (let index = start; index < swText.length; index += 1) {
+    const char = swText[index];
+    if (inString) {
+      if (char === '\\') index += 1;
+      else if (char === '"') inString = false;
+      continue;
+    }
+    if (char === '"') inString = true;
+    else if (char === '[') depth += 1;
+    else if (char === ']') {
+      depth -= 1;
+      if (depth === 0) return JSON.parse(swText.slice(start, index + 1));
+    }
+  }
+  return null;

+const TYPES = {
+  '.html': 'text/html; charset=utf-8', '.js': 'text/javascript; charset=utf-8', '.css': 'text/css; charset=utf-8',
+  '.json': 'application/json', '.webmanifest': 'application/manifest+json', '.wav': 'audio/wav', '.woff2': 'font/woff2', '.woff': 'font/woff',

+const NO_CACHE = new Set(['/sw.js', '/index.html', '/manifest.webmanifest']);
+
+export function startServer(dir, port) {
+  const root = path.resolve(dir);
+  const server = http.createServer(async (request, response) => {
+    const url = new URL(request.url ?? '/', 'http://localhost');
+    const pathname = decodeURIComponent(url.pathname) === '/' ? '/index.html' : decodeURIComponent(url.pathname);
+    const file = path.join(root, pathname);
+    if (request.method !== 'GET' && request.method !== 'HEAD') { response.writeHead(405).end(); return; }
+    if (!file.startsWith(root + path.sep)) { response.writeHead(403).end(); return; }
+    try {
+      const body = await fs.readFile(file);
+      response.writeHead(200, {
+        'Content-Type': TYPES[path.extname(file)] ?? 'application/octet-stream',
+        'Cache-Control': NO_CACHE.has(pathname) ? 'no-cache' : 'max-age=0'
+      });
+      response.end(request.method === 'HEAD' ? undefined : body);
+    } catch {
+      response.writeHead(404).end();
+    }
+  });
+  return new Promise((resolve, reject) => {
+    server.once('error', reject);
+    server.listen(port, '127.0.0.1', () => resolve(server));
+  });
