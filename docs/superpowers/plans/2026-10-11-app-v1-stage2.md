> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 生徒が「種目タイル → ▶」の2タップ（2回目以降は「前回」カードの1タップ）でカウントを鳴らせる S1・S2 を、iOS の音声解除・2秒判定・中断検知・WakeLock・PC 配置・高齢者向け数値基準の機械検査（新 `ui:check`）つきで作る。

**Tech Stack:** Node ≥22.18（検証は v24.13.0）、Vite 8.3.4、Preact 10.29.8、TypeScript 6.0.3、node:test、playwright-core（既存）、**新規** `@fontsource/inter` 5.3.0・`@fontsource/noto-sans-jp` 5.3.0。

## Review Focus

仕様が求めるが、人が踏みやすく、タスクの通常のテストだけでは見落としやすいもの（上ほど起きやすい）。各行のテストは担当 Task に入れてある。

3. **壊れた・古い・別の版の保存データ／プライベートモード**（IndexedDB が開けない、`countInBars: 2`、知らない種目、文字列の速さ）→ 項目ごとに既定へ戻す `normalizeSettings` と、失敗しても投げない `loadSettings`/`saveSettings`（Task 3 の `settings.test.ts`・`store.test.ts`）。
4. **▶ の二度押し・読み込み中の連打**（声の読み込み中にもう一度押す）→ 後の操作が勝ち、先の再生は始まらない（Task 7「a second ▶ while the first is still loading wins」）。
5. **オフラインで初めて ▶**（標準音声がまだ端末に無い）→ 鳴らさず「この声はまだ端末に入っていません…」を出し、WakeLock を解放（Task 7「a missing voice … says so and does not start」）。

```ts
//   document.defaultView・document の可視状態とイベント（visibilityState・hidden・add/removeEventListener）
// - 裸の識別子だけ禁止（obj.history / obj.addEventListener のような他人のプロパティは許す）: window・navigator・globalThis・self・
//   location・history・addEventListener・removeEventListener・requestAnimationFrame・cancelAnimationFrame・fetch
//   → `const history = []` のような同名ローカル変数も検出する（紛らわしいので別名にする方針）。

const ALWAYS = /\b(localStorage|sessionStorage|indexedDB|webkitAudioContext|matchMedia)\b|\bnew\s+(AudioContext|MediaRecorder|Audio)\b|\bdocument\s*\.\s*(defaultView|visibilityState|hidden|addEventListener|removeEventListener)\b/;
const BARE = /(?<![.\w$])(window|navigator|globalThis|self|location|history|addEventListener|removeEventListener|requestAnimationFrame|cancelAnimationFrame|fetch)\b/;
const IN_STRINGS = /\bcreateElement\s*\(\s*['"`]audio['"`]|['"`]visibilitychange['"`]/;

// コメントと文字列の中身を落とす（ブロックコメントは行をまたいで追う）。
export function findDeviceAccess(source: string): { line: number; text: string }[] {
  const found: { line: number; text: string }[] = [];
  let inBlock = false;
  source.split('\n').forEach((raw, index) => {
    let code = '';
    let rest = raw;
    while (rest.length > 0) {
      if (inBlock) {
        const close = rest.indexOf('*/');
        if (close < 0) { rest = ''; break; }
        inBlock = false;
        rest = rest.slice(close + 2);
        continue;
      }
      const open = rest.indexOf('/*');
      const lineComment = rest.search(/(^|\s)\/\//);
      if (lineComment >= 0 && (open < 0 || lineComment < open)) { code += rest.slice(0, lineComment); rest = ''; break; }
      if (open < 0) { code += rest; rest = ''; break; }
      code += rest.slice(0, open);
      inBlock = true;
      rest = rest.slice(open + 2);
    }
    const withStrings = code;
    // 文字列は空にするが、テンプレート文字列の ${...} の中身はコードとして残す（`${window.innerWidth}` の見逃し防止）。
    code = code.replace(/'(?:\\.|[^'\\])*'|"(?:\\.|[^"\\])*"|`(?:\\.|[^`\\])*`/g, literal =>
      literal.startsWith('`') ? ` "" ${[...literal.matchAll(/\$\{([^}]*)\}/g)].map(match => match[1]).join(' ; ')} "" ` : '""');
    if (ALWAYS.test(code) || BARE.test(code) || IN_STRINGS.test(withStrings)) found.push({ line: index + 1, text: raw.trim() });
  });
  return found;
}
```

```ts
// ハッシュ ↔ 画面の対応表（仕様 §3）。純粋関数だけ。window には触らない（platform/hash.ts の役目）。
export type ScreenId = 'S1' | 'S2' | 'S3' | 'S4' | 'S5' | 'S6' | 'S7' | 'S8' | 'S9' | 'S10' | 'S11' | 'S12' | 'S13';
export type ScreenKind = 'root' | 'focus' | 'list';
export type TabId = 'practice' | 'routines' | 'settings';
export type RouteParams = { danceId?: string; routineId?: string };
// 画面の見た目だけを強制するテスト用クエリ（Ruling 23・33）。設定・hints・再生状態には書かない。
export type UiState = 'no-sound' | 'wakelock-tip';
export type RouteQuery = { uiState?: UiState };
export interface Route { screen: ScreenId; params: RouteParams; query: RouteQuery }
export type RouteTarget = Pick<Route, 'screen' | 'params'>;

interface ScreenSpec { kind: ScreenKind; parent: ScreenId | null; tab: TabId; path: string }

// path の ":name" は ID 1区切り（英数字・ハイフン・下線のみ）。
export const SCREENS: Record<ScreenId, ScreenSpec> = {
  S1: { kind: 'root', parent: null, tab: 'practice', path: '/' },
  S2: { kind: 'focus', parent: 'S1', tab: 'practice', path: '/practice/:danceId' },
  S3: { kind: 'root', parent: null, tab: 'routines', path: '/routines' },
  S4: { kind: 'focus', parent: 'S3', tab: 'routines', path: '/routines/new' },
  S5: { kind: 'focus', parent: 'S3', tab: 'routines', path: '/routines/:routineId/play' },
  S6: { kind: 'root', parent: null, tab: 'settings', path: '/settings' },
  S7: { kind: 'list', parent: 'S6', tab: 'settings', path: '/voices' },
  S8: { kind: 'focus', parent: 'S7', tab: 'settings', path: '/voices/record' },
  S9: { kind: 'focus', parent: 'S7', tab: 'settings', path: '/share' },
  S10: { kind: 'focus', parent: 'S7', tab: 'settings', path: '/receive' },
  S11: { kind: 'focus', parent: 'S6', tab: 'settings', path: '/help/sound' },
  S12: { kind: 'list', parent: 'S6', tab: 'settings', path: '/help/install' },
  S13: { kind: 'list', parent: 'S6', tab: 'settings', path: '/about' }
};

// S4 は「新規」と「編集」の2つの形を持つので、編集の形をここで足す。
const EXTRA_PATHS: Array<{ screen: ScreenId; path: string }> = [{ screen: 'S4', path: '/routines/:routineId/edit' }];

const ID_SEGMENT = /^[A-Za-z0-9_-]+$/;
const UI_STATES: readonly UiState[] = ['no-sound', 'wakelock-tip'];

const PATTERNS = [
  ...(Object.entries(SCREENS) as Array<[ScreenId, ScreenSpec]>).map(([screen, spec]) => ({ screen, path: spec.path })),
  ...EXTRA_PATHS
].map(({ screen, path }) => ({ screen, segments: path.split('/').filter(Boolean) }));

export function parseHash(hash: string): Route {
  const [rawPath = '', rawQuery = ''] = hash.replace(/^#/, '').split('?');
  const path = rawPath;
  const query = parseQuery(rawQuery);
  const segments = path.split('/').filter(Boolean);
  for (const pattern of PATTERNS) {
    if (pattern.segments.length !== segments.length) continue;
    const params: Record<string, string> = {};
    const matched = pattern.segments.every((expected, index) => {
      const actual = segments[index] ?? '';
      if (expected.startsWith(':')) {
        if (!ID_SEGMENT.test(actual)) return false;
        params[expected.slice(1)] = actual;
        return true;
      }
      return expected === actual;
    });
    if (matched) return { screen: pattern.screen, params, query };
  }
  return { screen: 'S1', params: {}, query };
}

// 知らないキー・値は捨てる。
function parseQuery(raw: string): RouteQuery {
  const value = new URLSearchParams(raw).get('ui-state');
  const uiState = UI_STATES.find(state => state === value);
  return uiState ? { uiState } : {};
}

// クエリは出さない（遷移すると ui-state は消える）。
export function buildHash(route: RouteTarget): string {
  const params = route.params as Record<string, string | undefined>;
  const candidates = PATTERNS.filter(pattern => pattern.screen === route.screen);
  for (const pattern of candidates) {
    const names = pattern.segments.filter(segment => segment.startsWith(':')).map(segment => segment.slice(1));
    const given = Object.keys(params).filter(name => params[name] !== undefined);
    if (names.length !== given.length || !names.every(name => given.includes(name))) continue;
    const path = pattern.segments.map(segment => (segment.startsWith(':') ? params[segment.slice(1)] : segment)).join('/');
    return `#/${path}`;
  }
  throw new Error(`No hash for ${route.screen} with params ${JSON.stringify(route.params)}`);
}

export function showsTabs(screen: ScreenId): boolean {
  return SCREENS[screen].kind !== 'focus';
}

export function parentScreen(screen: ScreenId): ScreenId | null {
  return SCREENS[screen].parent;
}

export function tabOf(screen: ScreenId): TabId {
  return SCREENS[screen].tab;
}
```

```ts
// 「戻る」の判断に使う、このタブで見てきたハッシュの列（仕様 §3・Ruling 17）。純粋関数だけ。
// ひとつ前と同じハッシュに来たら「戻った」とみなして末尾を落とし、そうでなければ積む。
// 列が1件（直接開いた・再読み込みした）なら、戻る先は履歴ではなく親画面（置き換え）。
export function recordVisit(stack: readonly string[], hash: string): string[] {
  const visit = hashPath(hash);
  if (stack.length >= 2 && stack[stack.length - 2] === visit) return stack.slice(0, -1);
  if (stack[stack.length - 1] === visit) return [...stack];
  return [...stack, visit];
}

export function canGoBackInHistory(stack: readonly string[]): boolean {
  return stack.length > 1;
}

// クエリ（?ui-state=…）は画面の違いではないので比べない。空のハッシュはホーム。
function hashPath(hash: string): string {
  const path = hash.split('?')[0] ?? '';
  return path === '' || path === '#' ? '#/' : path;
}
```

```ts
import { useEffect, useState } from 'preact/hooks';
import { historyBack, onHashChange, readHash, setHash } from '../platform/hash.ts';
import { canGoBackInHistory, recordVisit } from '../router/back-stack.ts';
import { buildHash, parentScreen, parseHash, type Route, type RouteTarget, type ScreenId } from '../router/routes.ts';

// このタブで見てきた画面の列（「戻る」の判断用。仕様 §3・Ruling 17）。起動時の画面から始まる。
let visited: string[] = recordVisit([], readHash());

export function useRoute(): Route {
  const [route, setRoute] = useState<Route>(() => parseHash(readHash()));
  useEffect(() => onHashChange(hash => {
    visited = recordVisit(visited, hash);
    setRoute(parseHash(hash));
  }), []);
  return route;
}

export function navigate(target: RouteTarget): void {
  setHash(buildHash(target));
}

// 履歴があれば1つ戻る。直接開いた等で無ければ親画面へ置き換え（親は params を持たない画面だけ）。
export function goBack(current: ScreenId): void {
  if (canGoBackInHistory(visited)) {
    historyBack();
    return;
  }
  const parent = parentScreen(current) ?? 'S1';
  setHash(buildHash({ screen: parent, params: {} }), { replace: true });
}
```

```ts
export const BPM_MIN = 40;
export const BPM_MAX = 220;

export type SpeedPreset = 'slow' | 'standard' | 'fast';

export function speedPresetBpms(profile: SpeedProfile): Record<SpeedPreset, number> {
  return {
    slow: Math.round(profile.practiceBpm * SPEED_SLOW_RATIO),
    standard: profile.practiceBpm,
    fast: Math.min(Math.round(profile.practiceBpm * SPEED_FAST_RATIO), profile.competitionBpm.max)
  };
}

  if (!Number.isFinite(number)) return null;
  return Math.min(BPM_MAX, Math.max(BPM_MIN, Math.round(number)));
```

```ts
// 端末内の保存（仕様 §7・Ruling 11）。IndexedDB `dance-count-v2` を version 2 に上げ、
// 既存の voice-packs を残したまま routines と settings を足す。版上げはここ1か所だけで行う。
// 注意: src/engine/recorded-voice-store.js は version 1 を直接開くので、v2 にした後に呼ぶと VersionError。
// 新アプリからは段4まで呼ばない（段3で platform/storage 経由に作り直す）。
export const DB_NAME = 'dance-count-v2';
export const DB_VERSION = 2;
export type StoreName = 'routines' | 'settings' | 'voice-packs';

// keyPath を持つストアは値の id で、持たないストア（settings）は引数のキーで保存する。
const STORES: Record<StoreName, IDBObjectStoreParameters | undefined> = {
  'voice-packs': { keyPath: 'id' },
  routines: { keyPath: 'id' },
  settings: undefined
};

export interface UpgradableDatabase {
  objectStoreNames: { contains(name: string): boolean };
  createObjectStore(name: string, options?: IDBObjectStoreParameters): unknown;
}

// 版上げの中身（Ruling 26。テストは偽の db で行う）。既にあるストアには触らない（削除・改名もしない）。
// oldVersion は 0（新規）か 1（旧画面で録音した端末）。どちらでも3ストアが揃う。
export function upgradeAppDatabase(db: UpgradableDatabase, _oldVersion: number): void {
  for (const [name, options] of Object.entries(STORES) as Array<[StoreName, IDBObjectStoreParameters | undefined]>) {
    if (!db.objectStoreNames.contains(name)) db.createObjectStore(name, options);
  }
}

export interface AppStorage {
  get<T>(store: StoreName, key: string): Promise<T | undefined>;
  // keyPath を持つストア（voice-packs・routines）は値の id がキー。settings は key を渡す。
  put<T>(store: StoreName, value: T, key?: string): Promise<void>;
}

let opening: Promise<AppStorage> | null = null;

// 開けない（IndexedDB が無い・プライベートモードで拒否）ときは reject。呼び出し側は既定値で動き続ける。
export function openAppStorage(): Promise<AppStorage> {
  opening ??= openDatabase().then(wrap, error => {
    opening = null;
    throw error;
  });
  return opening;
}

function openDatabase(): Promise<IDBDatabase> {
  return new Promise((resolve, reject) => {
    if (typeof indexedDB === 'undefined') return reject(new Error('IndexedDB is not available'));
    const request = indexedDB.open(DB_NAME, DB_VERSION);
    request.onupgradeneeded = event => upgradeAppDatabase(request.result, event.oldVersion);
    request.onsuccess = () => resolve(request.result);
    request.onerror = () => reject(request.error ?? new Error('IndexedDB open failed'));
    request.onblocked = () => reject(new Error('IndexedDB upgrade is blocked by another tab'));
  });
}

function wrap(db: IDBDatabase): AppStorage {
  return {
    get<T>(store: StoreName, key: string) {
      return run<T | undefined>(db, store, 'readonly', objectStore => objectStore.get(key));
    },
    async put<T>(store: StoreName, value: T, key?: string) {
      await run(db, store, 'readwrite', objectStore => (objectStore.keyPath ? objectStore.put(value) : objectStore.put(value, key)));
    }
  };
}

function run<T>(db: IDBDatabase, store: StoreName, mode: IDBTransactionMode, action: (objectStore: IDBObjectStore) => IDBRequest): Promise<T> {
  return new Promise((resolve, reject) => {
    const request = action(db.transaction(store, mode).objectStore(store));
    request.onsuccess = () => resolve(request.result as T);
    request.onerror = () => reject(request.error ?? new Error(`IndexedDB ${mode} failed`));
  });
}
```

```ts
  const hints = isRecord(source.hints) ? source.hints : {};

    uiLocale: source.uiLocale === 'en' || source.uiLocale === 'ja-JP' ? source.uiLocale : null,
    voiceLocale: source.voiceLocale === 'ja-JP' ? 'ja-JP' : 'en',

    speedPresetByDance: pickByDance(source.speedPresetByDance, (_danceId, value) =>
      SPEED_CHOICES.find(choice => choice === value)),
    customBpmByDance: pickByDance(source.customBpmByDance, (_danceId, value) => clampBpm(value) ?? undefined),

    countInBars: source.countInBars === 0 ? 0 : 1,
    loop: typeof source.loop === 'boolean' ? source.loop : DEFAULT_SETTINGS.loop,
    hints: { wakeLockTipShown: hints.wakeLockTipShown === true }

function pickByDance<T>(value: unknown, accept: (danceId: string, item: unknown) => T | undefined): Record<string, T> {
  const result: Record<string, T> = {};
  if (!isRecord(value)) return result;
  for (const [danceId, item] of Object.entries(value)) {
    if (!Object.hasOwn(DANCES, danceId)) continue;
    const accepted = accept(danceId, item);
    if (accepted !== undefined) result[danceId] = accepted;
  }
  return result;
}

  return typeof value === 'object' && value !== null && !Array.isArray(value);
```

```ts
import type { AppStorage } from '../platform/storage.ts';
import { DEFAULT_SETTINGS, SETTINGS_KEY, normalizeSettings, type AppSettings } from './settings.ts';

// 設定の読み書き（Ruling 26）。settings ストアの1レコード（key 'app'、schemaVersion 1）。
// 書きは read-merge-write: 別タブや後の段が足した項目を消さない。失敗は投げずに false（設定は失って困らない）。
export const SETTINGS_SCHEMA_VERSION = 1;

export interface LegacyLocaleSource { read(): string | null; forget(): void }

export async function loadSettings(storage: AppStorage | null, legacy: LegacyLocaleSource): Promise<AppSettings> {
  let saved: unknown;
  try { saved = await storage?.get('settings', SETTINGS_KEY); } catch { saved = undefined; }
  const settings = normalizeSettings(saved);
  if (settings.uiLocale !== null) return settings;
  const legacyLocale = normalizeSettings({ uiLocale: legacy.read() }).uiLocale;
  if (legacyLocale === null) return settings;
  const migrated = { ...settings, uiLocale: legacyLocale };
  if (await saveSettings(storage, { uiLocale: legacyLocale })) legacy.forget();
  return migrated;
}

export async function saveSettings(storage: AppStorage | null, patch: Partial<AppSettings>): Promise<boolean> {
  if (!storage) return false;
  try {
    const current = await storage.get<Record<string, unknown>>('settings', SETTINGS_KEY);
    const base = current && typeof current === 'object' ? current : {};
    await storage.put('settings', { ...base, ...patch, schemaVersion: SETTINGS_SCHEMA_VERSION }, SETTINGS_KEY);
    return true;
  } catch {
    return false;
  }
}

export { DEFAULT_SETTINGS };
```

```tsx
      .then(opened => { storage.current = opened; return loadSettings(opened, { read: readLegacyUiLocale, forget: forgetLegacyUiLocale }); })
      .then(loaded => { if (alive) { current.current = loaded; setSettings(loaded); } });
    return () => { alive = false; };
  }, []);

  const update = useCallback((patch: Partial<AppSettings>) => {
    current.current = { ...current.current, ...patch };
    setSettings(current.current);
    void saveSettings(storage.current, patch);
  }, []);

  const locale: AppLocale = settings?.uiLocale ?? resolveAppLocale(deviceLanguages());
  const translator = useMemo(() => createAppTranslator(locale), [locale]);
  const i18n = useMemo(() => ({ translator, setLocale: (next: AppLocale) => update({ uiLocale: next }) }), [translator, update]);
  useEffect(() => applyDocumentLanguage(locale, translator.t('documentTitle')), [locale, translator]);

  if (!settings) return null;
  const screen = resolveScreen(route);
  const Simple = SIMPLE_SCREENS[screen] ?? HomeScreen;
  return (
    <I18nContext.Provider value={i18n}>
      <SettingsContext.Provider value={{ settings, update }}>
        <div class="dc-shell" lang={locale}>
          <Simple />
          {showsTabs(screen) && <TabBar current={tabOf(screen)} />}
        </div>
      </SettingsContext.Provider>
    </I18nContext.Provider>
  );
}
```

```ts
import { compileRhythm } from '../../engine/rhythm-compiler.js';

// 種目の副題・カウントの型・大きな現在カウントに共通の表示語（Ruling 22・28）。
// 声の言語・UI 言語に依存しない。count-vocabularies の prompts（録音用の表記）は使わない。拍子は出さない。
const DISPLAY_BY_VOICE_KEY: Record<string, string> = {
  one: '1', two: '2', three: '3', four: '4', five: '5', six: '6', seven: '7', eight: '8',
  slow: 'S', quick: 'Q', and: '&', ah: 'a'
};

export function displayForVoiceKey(voiceKey: string | null | undefined): string {
  return voiceKey && Object.hasOwn(DISPLAY_BY_VOICE_KEY, voiceKey) ? DISPLAY_BY_VOICE_KEY[voiceKey]! : '';
}

// カウントの型（例 '12&3'）→ '1 2 & 3'。声の無いトークン（@ 等）は表示しない。
export function formatCounts(danceId: string, counts: string): string {
  const { events } = compileRhythm({ danceId, counts }) as { events: Array<{ voiceKey: string | null }> };
  return events.map(event => displayForVoiceKey(event.voiceKey)).filter(Boolean).join(' ');
}
```

```ts
import { TICKS_PER_BEAT } from '../../engine/dance-profiles.js';

// 拍インジケータの「いま何拍目か」（0 始まり）。tick はエンジンの onFrame が返す originalTick だけを使う
// （時刻から拍を計算し直さない: 第2の時間計算を作らない）。tick が無い（停止中・はじめの合図の間）は null。
export function beatIndexInBar(originalTick: number | null | undefined, beatsPerBar: number): number | null {
  if (typeof originalTick !== 'number' || !Number.isFinite(originalTick) || !(beatsPerBar > 0)) return null;
  const barTicks = beatsPerBar * TICKS_PER_BEAT;
  const inBar = ((originalTick % barTicks) + barTicks) % barTicks;
  return Math.floor(inBar / TICKS_PER_BEAT);
}
```

```ts
import type { AppSettings } from '../settings/settings.ts';
import { SPEED_PRESETS, clampBpm, resolveBpm, speedPresetBpms, type SpeedChoice, type SpeedProfile } from './speed.ts';

// 種目ごとの記憶（仕様 §7）から S2 の現在の選択を作り、操作を設定の差分に直す（純粋関数）。
export interface DanceProfileLike extends SpeedProfile { id: string; templates: readonly string[] }

export interface PracticeChoice {
  counts: string;            // カウントの型（記憶が無ければ templates[0]。Ruling 18）
  speed: SpeedChoice;
  bpm: number;
  customBpm: number | null;  // 任意の速さのときだけ（「くわしく · ♩=値」。Ruling 31）
}

export function practiceChoice(profile: DanceProfileLike, settings: AppSettings): PracticeChoice {
  const remembered = settings.lastTemplateByDance[profile.id];
  const counts = remembered && profile.templates.includes(remembered) ? remembered : profile.templates[0]!;
  const speed = settings.speedPresetByDance[profile.id] ?? 'standard';
  const custom = settings.customBpmByDance[profile.id];
  const effective: SpeedChoice = speed === 'custom' && custom === undefined ? 'standard' : speed;
  const bpm = resolveBpm(profile, effective, custom);
  return { counts, speed: effective, bpm, customBpm: effective === 'custom' ? bpm : null };
}

export function choosePreset(settings: AppSettings, danceId: string, preset: Exclude<SpeedChoice, 'custom'>): Partial<AppSettings> {
  const { [danceId]: _removed, ...customBpmByDance } = settings.customBpmByDance;
  return { speedPresetByDance: { ...settings.speedPresetByDance, [danceId]: preset }, customBpmByDance };
}

// 任意の値。3択のどれかと同じ値ならそのプリセットにする（Ruling 31 の追加規則）。
export function chooseCustomBpm(settings: AppSettings, profile: DanceProfileLike, value: unknown): Partial<AppSettings> {
  const bpm = clampBpm(value);
  if (bpm === null) return {};
  const presets = speedPresetBpms(profile);
  const same = SPEED_PRESETS.find(preset => presets[preset] === bpm);
  if (same) return choosePreset(settings, profile.id, same);
  return {
    speedPresetByDance: { ...settings.speedPresetByDance, [profile.id]: 'custom' },
    customBpmByDance: { ...settings.customBpmByDance, [profile.id]: bpm }
  };
}

export function chooseTemplate(settings: AppSettings, profile: DanceProfileLike, counts: string): Partial<AppSettings> {
  if (!profile.templates.includes(counts)) return {};
  return { lastTemplateByDance: { ...settings.lastTemplateByDance, [profile.id]: counts } };
}
```

```ts
// 再生中の画面スリープ防止（仕様 §9）。▶ のクリック処理の中で await より前に呼ぶこと
// （request は最初の await より前に同期で発行される）。画面が裏から戻ったら取り直し、停止時に解放する。
export interface ScreenWakeLock { release(): Promise<void> }

export async function keepScreenOn(): Promise<ScreenWakeLock | null> {
  if (!('wakeLock' in navigator)) return null;
  let sentinel: WakeLockSentinel | null = null;
  let released = false;
  const acquire = async () => {
    try { sentinel = await navigator.wakeLock.request('screen'); } catch { sentinel = null; }
  };
  const onVisibility = () => {
    if (!released && document.visibilityState === 'visible' && (!sentinel || sentinel.released)) void acquire();
  };
  await acquire();
  if (!sentinel) return null;
  document.addEventListener('visibilitychange', onVisibility);
  return {
    async release() {
      released = true;
      document.removeEventListener('visibilitychange', onVisibility);
      const current: WakeLockSentinel | null = sentinel;
      sentinel = null;
      await current?.release().catch(() => {});
    }
  };
}
```

```ts
  createEngine(audio: A, voicePack: V, onFrame: (frame: EngineFrame) => void): EngineLike;
  keepScreenOn(): Promise<WakeLockLike | null>;                             // クリック処理の中で同期に発行
  onStateChange(audio: A, listener: (state: string) => void): () => void;
  setTimer(callback: () => void, ms: number): () => void;                  // 戻り値で取り消し
}

export interface SessionHints {
  wakeTipAlreadyShown(): boolean;
  markWakeTipShown(): void;
}

export const IDLE_VIEW: SessionView = Object.freeze({
  playing: false, starting: false, countIn: false, voiceKey: null, originalTick: null, notice: 'none'
}) as SessionView;

export class PracticeSession<A extends AudioLike, V> {
  private readonly deps: SessionDeps<A, V>;
  private readonly hints: SessionHints;
  private view: SessionView = IDLE_VIEW;
  private readonly listeners = new Set<(view: SessionView) => void>();
  private audio: A | null = null;
  private engine: EngineLike | null = null;
  private wakeLock: WakeLockLike | null = null;
  private cancelSoundCheck: (() => void) | null = null;
  private unsubscribeState: (() => void) | null = null;
  private generation = 0;

  constructor(deps: SessionDeps<A, V>, hints: SessionHints) {
    this.deps = deps;
    this.hints = hints;
  }

  get current(): SessionView { return this.view; }

  subscribe(listener: (view: SessionView) => void): () => void {
    this.listeners.add(listener);
    listener(this.view);
    return () => { this.listeners.delete(listener); };
  }

  // ▶ のクリック処理から同期で呼ぶ。unlock と WakeLock の request は最初の await より前に発行される。
  play(options: PracticeOptions): Promise<void> {
    if (options.track !== null) throw new Error('Playing with a track is not supported in v1');
    if (typeof options.bpm !== 'number' || !Number.isFinite(options.bpm)) throw new TypeError('bpm must be a finite number');
    this.halt();
    const generation = ++this.generation;
    const audio = this.deps.unlock();
    this.attach(audio);
    const wakeLockRequest = this.deps.keepScreenOn();
    this.update({ ...IDLE_VIEW, starting: true });
    return this.begin(generation, audio, options, wakeLockRequest);
  }

  // ■・戻る・画面を離れる。出ている案内はそのまま（no-sound のバナーは次の ▶ で消える）。
  stop(): void {
    this.generation += 1;
    this.halt();
    this.update({ ...IDLE_VIEW, notice: this.view.notice });
  }

  dismissNotice(): void {
    this.update({ ...this.view, notice: 'none' });
  }

  dispose(): void {
    this.stop();
    this.unsubscribeState?.();
    this.unsubscribeState = null;
    this.listeners.clear();
  }

  private async begin(generation: number, audio: A, options: PracticeOptions, wakeLockRequest: Promise<WakeLockLike | null>) {
    const lockPromise = wakeLockRequest.catch(() => null);
    let voicePack: V;
    try {
      voicePack = await this.deps.loadVoice(audio);
    } catch {
      void lockPromise.then(lock => lock?.release());
      if (generation === this.generation) this.update({ ...IDLE_VIEW, notice: 'voice-missing' });
      return;
    }
    if (generation !== this.generation) {
      void lockPromise.then(lock => lock?.release());
      return;
    }
    this.engine ??= this.deps.createEngine(audio, voicePack, frame => this.onFrame(frame));
    const startedAt = audio.currentTime;
    this.engine.start({
      compiled: compileRhythm({ danceId: options.danceId, counts: options.counts }),
      bpm: options.bpm,
      loop: options.loop,
      metronome: options.metronome,
      countInBars: options.countInBars
    });
    this.update({ ...IDLE_VIEW, playing: true, countIn: options.countInBars > 0 });
    const lock = await lockPromise;
    if (generation !== this.generation) {
      void lock?.release();
      return;
    }
    this.wakeLock = lock;
    this.cancelSoundCheck = this.deps.setTimer(() => this.checkSound(generation, audio, startedAt, lock === null), SOUND_CHECK_DELAY_MS);
  }

  // 2秒判定（Ruling 6-2）。鳴っていれば、WakeLock が取れなかった時の案内を1回だけ出す（Ruling 30）。
  private checkSound(generation: number, audio: A, startedAt: number, wakeLockUnavailable: boolean) {
    if (generation !== this.generation || !this.view.playing) return;
    if (audio.state !== 'running' || !(audio.currentTime > startedAt)) {
      this.update({ ...this.view, notice: 'no-sound' });
      return;
    }
    if (wakeLockUnavailable && this.view.notice === 'none' && !this.hints.wakeTipAlreadyShown()) {
      this.hints.markWakeTipShown();
      this.update({ ...this.view, notice: 'wake-tip' });
    }
  }

  // 中断（電話・ロック）。再生中のときだけ止めて1行を出す。running に戻っても自動再開しない（Ruling 30）。
  private attach(audio: A) {
    if (this.audio === audio) return;
    this.unsubscribeState?.();
    this.audio = audio;
    this.engine = null;
    this.unsubscribeState = this.deps.onStateChange(audio, state => {
      if ((state === 'suspended' || state === 'interrupted') && this.view.playing) {
        this.generation += 1;
        this.halt();
        this.update({ ...IDLE_VIEW, notice: this.view.notice === 'no-sound' ? 'no-sound' : 'stopped' });
      }
    });
  }

  private onFrame(frame: EngineFrame) {
    if (!frame.playing || !this.view.playing) return;
    const countIn = frame.countIn === true;
    this.update({
      ...this.view,
      countIn,
      voiceKey: countIn ? null : frame.event?.voiceKey ?? null,
      originalTick: countIn ? null : frame.originalTick ?? null
    });
  }

  private halt() {
    this.engine?.stop();
    this.cancelSoundCheck?.();
    this.cancelSoundCheck = null;
    const lock = this.wakeLock;
    this.wakeLock = null;
    void lock?.release();
  }

  private update(view: SessionView) {
    this.view = view;
    for (const listener of this.listeners) listener(view);
  }
}
```

```tsx
import { SoundHelpScreen } from './screens/SoundHelpScreen.tsx';
import { VoicesScreen } from './screens/VoicesScreen.tsx';
import { DEFAULT_SETTINGS, type AppSettings } from './settings/settings.ts';
import { loadSettings, saveSettings } from './settings/store.ts';
import { I18nContext } from './state/i18n.ts';
import { useRoute } from './state/route.ts';
import { SettingsContext } from './state/settings.ts';

// 中身がある画面だけ登録する。未実装の画面はホームを出す（段3以降でここに足す）。
const SIMPLE_SCREENS: Partial<Record<ScreenId, ComponentType>> = {
  S1: HomeScreen,
  S3: RoutinesScreen,
  S6: SettingsScreen,
  S7: VoicesScreen,
  S11: SoundHelpScreen
};

// S2 は種目が分かる時だけ。知らない種目・未実装の画面はホーム。
export function resolveScreen(route: Route): ScreenId {
  if (route.screen === 'S2') return route.params.danceId && Object.hasOwn(DANCE_PROFILES, route.params.danceId) ? 'S2' : 'S1';
  return SIMPLE_SCREENS[route.screen] ? route.screen : 'S1';
}

export function App() {
  const route = useRoute();
  const [settings, setSettings] = useState<AppSettings | null>(null);
  const storage = useRef<AppStorage | null>(null);
  const current = useRef<AppSettings>(DEFAULT_SETTINGS);

  // 設定を読み終えてから描画する（前回カードのちらつき防止。Ruling 26）。開けなければ既定値で続行。
  useEffect(() => {
    let alive = true;
    openAppStorage()
      .catch(() => null)
      .then(opened => { storage.current = opened; return loadSettings(opened, { read: readLegacyUiLocale, forget: forgetLegacyUiLocale }); })
      .then(loaded => { if (alive) { current.current = loaded; setSettings(loaded); } });
    return () => { alive = false; };
  }, []);

  const update = useCallback((patch: Partial<AppSettings>) => {
    current.current = { ...current.current, ...patch };
    setSettings(current.current);
    void saveSettings(storage.current, patch);
  }, []);

  const locale: AppLocale = settings?.uiLocale ?? resolveAppLocale(deviceLanguages());
  const translator = useMemo(() => createAppTranslator(locale), [locale]);
  const i18n = useMemo(() => ({ translator, setLocale: (next: AppLocale) => update({ uiLocale: next }) }), [translator, update]);
  useEffect(() => applyDocumentLanguage(locale, translator.t('documentTitle')), [locale, translator]);

  if (!settings) return null;
  const screen = resolveScreen(route);
  const Simple = SIMPLE_SCREENS[screen] ?? HomeScreen;
  return (
    <I18nContext.Provider value={i18n}>
      <SettingsContext.Provider value={{ settings, update }}>
        <div class="dc-shell" lang={locale}>
          <IconSprite />
          {screen === 'S2'
            ? <PracticeScreen key={route.params.danceId} danceId={route.params.danceId!} uiState={route.query.uiState} />
            : <Simple />}
          {showsTabs(screen) && <TabBar current={tabOf(screen)} />}
        </div>
      </SettingsContext.Provider>
    </I18nContext.Provider>
  );
}
```

```js
// ---- ここから下はブラウザの中で動く（page.evaluate に関数ごと渡す。外の変数は使えない） ----
function audit({ state, LIMITS, reducedMotion }) {
  const issues = [];
  const TARGETS = 'button, a[href], select, summary, input:not([type="hidden"]), label[for], [role=button], [role=switch], [role=radio]';
  const OPS = 'button, a[href], [role=button], [role=switch], [role=radio], input:not([type="hidden"]), select, summary, label[for]';
  const EXCLUDED_OPS = '[data-nav], [data-back], [data-utility], summary[data-more], [data-alert-action]';
  const FIXED_BARS = '[data-nav], .dc-dock';

  const visible = element => {
    if (element.closest('[hidden]')) return false;
    for (let node = element; node; node = node.parentElement) {
      const style = getComputedStyle(node);
      if (style.display === 'none' || style.visibility === 'hidden') return false;
    }
    // 閉じた <details> の中身は見えない（summary だけが見える）。Chrome は中身の rect を返すことがあるので明示的に除く
    const closed = element.closest('details:not([open])');
    if (closed && !closed.querySelector(':scope > summary')?.contains(element)) return false;
    const rect = element.getBoundingClientRect();
    return rect.width > 0 && rect.height > 0;
  };
  const describe = element => `${element.tagName.toLowerCase()}${element.className && typeof element.className === 'string' ? `.${element.className.trim().split(/\s+/).join('.')}` : ''} "${(element.textContent ?? '').trim().replace(/\s+/g, ' ').slice(0, 24)}"`;
  const inFixedBar = element => Boolean(element.closest(FIXED_BARS));

  // 固定バーを除いた「見える範囲」（U4・U12・Ruling 21/23 の判定に使う）
  let usableTop = 0;
  let usableBottom = innerHeight;
  for (const bar of document.querySelectorAll(FIXED_BARS)) {
    if (!visible(bar)) continue;
    const rect = bar.getBoundingClientRect();
    if (rect.top <= 1) usableTop = Math.max(usableTop, rect.bottom);
    else usableBottom = Math.min(usableBottom, rect.top);
  }
  const fullyInView = element => {
    const rect = element.getBoundingClientRect();
    const top = inFixedBar(element) ? 0 : usableTop;
    const bottom = inFixedBar(element) ? innerHeight : usableBottom;
    return rect.top >= top - 0.5 && rect.bottom <= bottom + 0.5 && rect.left >= -0.5 && rect.right <= innerWidth + 0.5;
  };

  // U1 文字の大きさ
  const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
  const textElements = new Set();
  while (walker.nextNode()) {
    const parent = walker.currentNode.parentElement;
    if (parent && walker.currentNode.textContent.trim() && visible(parent) && !parent.closest('[aria-hidden="true"]:not(.dc-now)')) textElements.add(parent);
  }
  for (const element of textElements) {
    const size = parseFloat(getComputedStyle(element).fontSize);
    const fineprint = element.closest('[data-fineprint]');
    if (size < LIMITS.fineprintFont - 0.01) issues.push(`U1 font ${size}px < ${LIMITS.fineprintFont} ${describe(element)}`);
    else if (!fineprint && size < LIMITS.minFont - 0.01) issues.push(`U1 font ${size}px < ${LIMITS.minFont} ${describe(element)}`);
    else if (element.closest('p, li, [data-body]') && size < LIMITS.bodyFont - 0.01) issues.push(`U1 body font ${size}px < ${LIMITS.bodyFont} ${describe(element)}`);
  }

  // U2 主要ボタン
  for (const element of document.querySelectorAll('.dc-primary')) {
    if (!visible(element)) continue;
    const rect = element.getBoundingClientRect();
    if (rect.height < LIMITS.primaryHeight - 0.5) issues.push(`U2 primary height ${Math.round(rect.height)} < ${LIMITS.primaryHeight} ${describe(element)}`);
  }
  for (const element of document.querySelectorAll('[data-play]')) {
    if (!visible(element)) continue;
    const rect = element.getBoundingClientRect();
    if (rect.height < LIMITS.playHeight - 0.5) issues.push(`U2 play height ${Math.round(rect.height)} < ${LIMITS.playHeight}`);
    if (rect.width < innerWidth * LIMITS.playWidthRatio - 0.5) issues.push(`U2 play width ${Math.round(rect.width)} < ${Math.round(innerWidth * LIMITS.playWidthRatio)}`);
  }

  // U3 タップ対象の大きさと間隔（固定バー同士・固定バーと本文の間は対象外: スクロールで重なり方が変わるため）
  const targets = [...document.querySelectorAll(TARGETS)].filter(visible).filter(element => !(element.tagName === 'INPUT' && element.closest('label')));
  for (const element of targets) {
    const rect = element.getBoundingClientRect();
    if (rect.width < LIMITS.tap - 0.5 || rect.height < LIMITS.tap - 0.5) issues.push(`U3 tap ${Math.round(rect.width)}x${Math.round(rect.height)} ${describe(element)}`);
  }
  for (let i = 0; i < targets.length; i += 1) {
    for (let j = i + 1; j < targets.length; j += 1) {
      const a = targets[i];
      const b = targets[j];
      if (a.contains(b) || b.contains(a) || inFixedBar(a) || inFixedBar(b)) continue;
      const ra = a.getBoundingClientRect();
      const rb = b.getBoundingClientRect();
      const dx = Math.max(0, Math.max(ra.left, rb.left) - Math.min(ra.right, rb.right));
      const dy = Math.max(0, Math.max(ra.top, rb.top) - Math.min(ra.bottom, rb.bottom));
      const gap = Math.hypot(dx, dy);
      if (gap < LIMITS.tapGap - 0.5) issues.push(`U3 gap ${gap.toFixed(1)}px < ${LIMITS.tapGap} between ${describe(a)} and ${describe(b)}`);
    }
  }

  // U4 主要操作数（Ruling 18）: 同じ [data-group] は1つ、除外の要素は数えない
  const operations = scope => {
    const keys = new Set();
    for (const element of document.querySelectorAll(OPS)) {
      if (!visible(element) || element.closest(EXCLUDED_OPS)) continue;
      if (scope === 'initial' && !fullyInView(element)) continue;
      keys.add(element.closest('[data-group]') ?? element);
    }
    return keys.size;
  };
  const initialOps = operations('initial');
  const totalOps = operations('all');
  // 「初期表示」は画面を開いた直後のこと。「くわしく」を開いた状態は人の操作の後なので全体（9以下）だけを見る（Ruling 31: 開くと8）。
  if (!state.openMore && initialOps > LIMITS.initialOps) issues.push(`U4 initial operations ${initialOps} > ${LIMITS.initialOps}`);
  if (totalOps > LIMITS.totalOps) issues.push(`U4 total operations ${totalOps} > ${LIMITS.totalOps}`);
  if (state.ops !== undefined && initialOps !== state.ops) issues.push(`U4 initial operations ${initialOps}, expected ${state.ops}`);

  // U5 選択肢の数・U6「くわしく」は1つ
  for (const group of document.querySelectorAll('[data-group]')) {
    const choices = [...group.querySelectorAll(OPS)].filter(visible).length;
    if (choices > LIMITS.groupChoices) issues.push(`U5 group "${group.dataset.group}" has ${choices} choices > ${LIMITS.groupChoices}`);
  }
  for (const select of document.querySelectorAll('select')) {
    if (select.options.length > LIMITS.groupChoices) issues.push(`U5 select has ${select.options.length} options`);
  }
  const summaries = document.querySelectorAll('summary[data-more]').length;
  if (summaries > LIMITS.moreSummaries) issues.push(`U6 ${summaries} "more" sections > ${LIMITS.moreSummaries}`);

  // U7 コントラスト（文字・操作の枠線）
  const linear = channel => { const c = channel / 255; return c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4; };
  const luminance = rgb => 0.2126 * linear(rgb[0]) + 0.7152 * linear(rgb[1]) + 0.0722 * linear(rgb[2]);
  const parse = value => { const m = value.match(/rgba?\(([\d.]+),\s*([\d.]+),\s*([\d.]+)(?:,\s*([\d.]+))?\)/); return m ? { rgb: [+m[1], +m[2], +m[3]], alpha: m[4] === undefined ? 1 : +m[4] } : null; };
  const background = element => {
    for (let node = element; node; node = node.parentElement) {
      const parsed = parse(getComputedStyle(node).backgroundColor);
      if (parsed && parsed.alpha >= 0.9) return parsed.rgb;
    }
    return [255, 255, 255];
  };
  const ratio = (a, b) => { const la = luminance(a); const lb = luminance(b); return (Math.max(la, lb) + 0.05) / (Math.min(la, lb) + 0.05); };
  for (const element of textElements) {
    const style = getComputedStyle(element);
    const color = parse(style.color);
    if (!color) continue;
    const size = parseFloat(style.fontSize);
    const large = size >= 24 && Number(style.fontWeight) >= 700;
    const needed = large ? LIMITS.contrastLarge : LIMITS.contrastText;
    const value = ratio(color.rgb, background(element));
    if (value < needed) issues.push(`U7 text contrast ${value.toFixed(2)} < ${needed} ${describe(element)}`);
  }
  for (const element of targets) {
    const style = getComputedStyle(element);
    const width = parseFloat(style.borderTopWidth);
    if (!(width > 0) || style.borderTopStyle === 'none') continue;
    const border = parse(style.borderTopColor);
    if (!border || border.alpha < 0.9) continue;
    const value = ratio(border.rgb, background(element.parentElement ?? element));
    if (value < LIMITS.contrastUi) issues.push(`U7 border contrast ${value.toFixed(2)} < ${LIMITS.contrastUi} ${describe(element)}`);
  }

  // U8 動き
  const durations = value => value.split(',').map(part => parseFloat(part) * (part.trim().endsWith('ms') ? 1 : 1000));
  for (const element of document.querySelectorAll('*')) {
    const style = getComputedStyle(element);
    for (const ms of [...durations(style.transitionDuration), ...durations(style.animationDuration)]) {
      if (reducedMotion ? ms > 0 : ms > LIMITS.motionMs) { issues.push(`U8 motion ${ms}ms ${describe(element)}`); break; }
    }
  }

  // U10 はみ出し・ボタン文言は2行まで
  for (const element of textElements) {
    if (element.scrollWidth > element.clientWidth + 1 && getComputedStyle(element).overflowX !== 'visible') {
      issues.push(`U10 clipped text ${element.scrollWidth} > ${element.clientWidth} ${describe(element)}`);
    }
  }
  // ボタン文言の行数は、文字列（テキストノード）ごとの行の数で数える（アイコンや2段組みの配置は行に数えない）
  for (const button of targets) {
    if (!['BUTTON', 'A', 'SUMMARY'].includes(button.tagName)) continue;
    const texts = document.createTreeWalker(button, NodeFilter.SHOW_TEXT);
    while (texts.nextNode()) {
      if (!texts.currentNode.textContent.trim()) continue;
      const range = document.createRange();
      range.selectNodeContents(texts.currentNode);
      const lines = new Set([...range.getClientRects()].filter(rect => rect.width > 0).map(rect => Math.round(rect.top))).size;
      if (lines > LIMITS.buttonLines) { issues.push(`U10 button text ${lines} lines > ${LIMITS.buttonLines} ${describe(button)}`); break; }
    }
  }

  // U11 下タブと戻る（種別の表と照合）
  const nav = document.querySelector('[data-nav]');
  const back = document.querySelector('[data-back]');
  const navVisible = Boolean(nav && visible(nav));
  const backVisible = Boolean(back && visible(back));
  if (state.kind === 'focus' && navVisible) issues.push('U11 tabs are visible on a focus screen');
  if (state.kind !== 'focus' && !navVisible) issues.push('U11 tabs are missing on a root/list screen');
  if (state.kind === 'root' && backVisible) issues.push('U11 back button on a root screen');
  if (state.kind !== 'root') {
    if (!backVisible) issues.push('U11 back button missing');
    else {
      const rect = back.getBoundingClientRect();
      if (rect.height < LIMITS.tap - 0.5 || rect.width < LIMITS.tap - 0.5) issues.push('U11 back button smaller than 48px');
      if (!back.textContent.trim()) issues.push('U11 back button has no text');
    }
  }

  // U12 S1: 先生カードと10種目が初期表示に入る（Ruling 16）
  if (state.teacherAndTiles) {
    const teacher = document.querySelector('[data-teacher-card]');
    if (!teacher || !fullyInView(teacher)) issues.push('U12 teacher card is not fully in the first view');
    const tiles = [...document.querySelectorAll('[data-group="dance"] > *')];
    if (tiles.length !== 10) issues.push(`U12 expected 10 dance tiles, found ${tiles.length}`);
    tiles.forEach((tile, index) => { if (!fullyInView(tile)) issues.push(`U12 dance tile ${index + 1} is not fully in the first view (bottom ${Math.round(tile.getBoundingClientRect().bottom)} > ${Math.round(usableBottom)})`); });
  }

  // U13 固定バーが画面の端に接している（下タブ・再生帯。PC の上部タブは上端）
  for (const bar of document.querySelectorAll(FIXED_BARS)) {
    if (!visible(bar)) continue;
    const rect = bar.getBoundingClientRect();
    const touches = Math.abs(rect.bottom - innerHeight) <= 2 || Math.abs(rect.top) <= 2;
    if (!touches) issues.push(`U13 fixed bar not at an edge (top ${Math.round(rect.top)} bottom ${Math.round(rect.bottom)}) ${describe(bar)}`);
  }

  // Ruling 21・23: 速さと ▶ は初期表示に入る。警告の枠は 375 幅で 120px 以下、中の操作は1つ
  if (state.speedAndPlay) {
    for (const selector of ['[data-group="speed"]', '[data-play]']) {
      const element = document.querySelector(selector);
      if (!element || !fullyInView(element)) issues.push(`R21/23 ${selector} is not fully in the first view`);
    }
  }
  if (state.alert) {
    const alert = document.querySelector('[data-alert]');
    if (!alert || !visible(alert)) issues.push('R23 alert is not shown');
    else {
      if (innerWidth < 1024 && alert.getBoundingClientRect().height > LIMITS.alertHeight + 0.5) issues.push(`R23 alert height ${Math.round(alert.getBoundingClientRect().height)} > ${LIMITS.alertHeight}`);
      const actions = [...alert.querySelectorAll(OPS)].filter(visible);
      if (actions.length > 1 || actions.some(action => !action.matches('[data-alert-action]'))) issues.push('R23 alert has an action other than one [data-alert-action]');
    }
  }
  return issues;
}
```
