## 計画者の発見（裁定の範囲で解決済み。台帳に記録する）

1. **`withRecordingSession` は必須**（実証 3）。`recordTake`・`requestMicrophone` が自分で通す。`run()` は同期で呼ぶ（`getUserMedia` をタップの処理の中・最初の await より前に発行するため）。
2. **モノにするのは正規化の前**（Ruling 58-3 の「`createMonoBuffer` を platform に」の読み替え）: `processTake` は受け取った AudioContext の `createBuffer(1, …)` にモノを写してから `normalizeAudioBuffer` に渡す。結果は同じで、platform に関数を足さずに Node でテストできる。
3. **波形は WAV のバイト列から直接作る**（`voices/wave.ts`）。AudioContext はタップの中でしか作らない（iOS）ので、語を移った時・再読み込みの後にも描けるように、自分で書いた 16bit WAV を読む。
4. **声のタイミングを変えても `updatedAt` を進めない**（decode の記憶の鍵＝Ruling 56-7。音は変わらない）。語を録った時だけ進める。
5. **`isCompletePack` は型の絞り込み（`pack is …`）にしない**（`packs.filter(p => !isCompletePack(p))` が `never[]` になる）。
6. **S8 は `ui-state` ごとに作り直す**（`key={route.query.uiState ?? ''}`）。同じ画面のままハッシュのクエリだけが変わると、前の状態（語の画面）が残る。
7. **入力欄の見出しを `<label for>` にすると ui:check がタップ対象（48px）として数え、U3 で落ちる** → `div`＋`aria-labelledby`（S4 の名前欄と同じ）。
8. **S7 の開いた行の操作は上に 8px の余白**（行のボタンとの間隔。U3）。
9. **前回カード・S3 の ▶ からの即再生は、声が決まってから**（Ruling 61-6 の読み方）: 録音の声は端末から読むので、マウント直後は `null`。即再生の effect を「声が決まった時点で 1 回」にした（数十ミリ秒の差）。▶ を押した時に未決なら標準（英語）で鳴る。声チップも決まるまでは標準の名前（1 描画ぶん）。
10. **聞き比べの確かめ方**: S8 には現在カウントの表示が無いので、スモークは「押したボタンが `aria-pressed="true"` になり、2 小節で自分で `false` に戻る」で鳴ったことを見る。
11. **`smoke-practice.js` の `expectSounding` は import できない**（読み込んだ時点で本体が走る）→ `smoke-voice.js` に同じ関数を持つ（`smoke-routines.js` と同じやり方）。Ruling 63-5 の「共有」はこの読み替え。
12. **`tools/make-fake-mic-wav.js` の「直接実行された時だけ書く」判定は `import.meta.filename`**（リポジトリのパスに空白があり、`file://${argv[1]}` とは一致しない）。
13. **S8 の語の画面の枠（声が入っていない・保存できない）は、説明の 1 行（「ボタンを押すと…」）の場所に出す**（枠と波形が同時に出ても 375 に収まるように。説明は 1 回目で役目を終えている）。
14. **320 幅の横スクロールを `ui:check` の検査にした**（ひとみうさの要件。全状態の初期表示・1 件の報告）。
15. 録音中は「録音する」を `disabled`（Ruling 59-2）。「‹ 戻る」と「前の語」は押せて、結果は `generation` で捨てる（Ruling 57-7）。
16. 段4で `standardVoiceJapanese`・`standardVoiceJapaneseShort` の文言キーを先に足す（`voiceDisplayName` が日本語の標準の声の名前を引けるように。Ruling 68-4 は 6c で足すとしている＝6c は JSON に 1 行足すだけで済む）。→ Open question 3。

```ts
export function tone({ totalMs = 1500, startMs = 300, lengthMs = 600, amplitude = 0.5, noise = 0, sampleRate = 48000 } = {}): Float32Array {

  for (let i = 0; i < samples.length; i += 1) samples[i] = noise * (i % 2 === 0 ? 1 : -1);
  const start = Math.round(startMs * sampleRate / 1000);
  const end = Math.min(samples.length, start + Math.round(lengthMs * sampleRate / 1000));

  return samples;
```

```ts
  en: 'user-recorded-v1',
  'ja-JP': 'user-recorded-ja-v1'
});

const VOICE_ID = /^[A-Za-z0-9_-]{1,64}$/;

export function isVoiceId(value: unknown): value is string {
  return typeof value === 'string' && VOICE_ID.test(value);

export function recordedPackIdFor(locale: VoiceLocale): string {
```

```ts
import { REQUIRED_VOICE_KEYS, createRecordedVoicePack, recordedPackProgress, validateRecordedVoicePack } from '../../engine/recorded-voice-pack.js';
import type { AppStorage } from '../platform/storage.ts';
import { isVoiceId, recordedPackIdFor, toVoiceLocale, type VoiceLocale } from './ids.ts';

// 声の保存（Ruling 56）。IndexedDB `dance-count-v2` v2 の voice-packs ストアに platform/storage 経由で読み書きする。
// エンジンの recorded-voice-store.js（v1 を直接開く）は使わない。形の検査はエンジンの validateRecordedVoicePack。
export const VOICE_LIMITS = Object.freeze({
  packs: 20,                       // 声の数（受け取りが増える段5で効く）
  nameLength: 40,                  // 名前（ルーティンと同じ）
  wordBytes: 2 * 1024 * 1024,      // 1 語の WAV（録音は 1 語 ≒ 70KB）
  packBytes: 20 * 1024 * 1024      // 1 つの声の合計
});

export type VoiceKey = 'slow' | 'quick' | 'one' | 'two' | 'three' | 'four' | 'five' | 'six' | 'seven' | 'eight' | 'and' | 'ah';
// S8 の語の順（仕様 §4 S8: スロー・クイック・1〜8・アンド・ア）
export const VOICE_KEYS = REQUIRED_VOICE_KEYS as readonly VoiceKey[];

export interface RecordedWord {
  blob: Blob;
  mimeType: string;
  onsetMs: number;
  sourceOnsetMs: number;
  durationMs: number;
  normalized: true;
  gainDb: number;
}

export interface VoiceTimingAdjustments { globalOffsetMs: number; wordOffsetsMs: Record<string, number>; adjustedAt: string }

export interface VoicePackRecord {
  schemaVersion: 1;
  type: 'dance-count-recorded-voice-pack';
  id: string;
  name: string;
  locale: VoiceLocale;
  vocabularyId: VoiceLocale;
  updatedAt: string;                       // 語を録るたびに更新（再生用に decode した声の記憶の鍵。Ruling 56-7）
  words: Partial<Record<VoiceKey, RecordedWord>>;
  timingAdjustments?: VoiceTimingAdjustments;
  origin: 'recorded' | 'received';
  distributor?: { name: string };          // 受け取った声（段5が書く）
  consent?: { voiceDistribution: true; consentedAt: string };   // 段5が書く
}

// 保存されていた値を信用しない。形が外れた物は null（一覧に出さない・消さない。段3の normalizeRoutine と同じ）。
export function normalizeVoicePackRecord(raw: unknown): VoicePackRecord | null {
  if (!isRecord(raw)) return null;
  try { validateRecordedVoicePack(raw, { requireComplete: false }); } catch { return null; }
  if (!isVoiceId(raw.id)) return null;
  if (typeof raw.name !== 'string' || raw.name.trim().length === 0 || [...raw.name].length > VOICE_LIMITS.nameLength) return null;
  const words = raw.words as Record<string, { blob: Blob }>;
  let total = 0;
  for (const word of Object.values(words)) {
    if (word.blob.size > VOICE_LIMITS.wordBytes) return null;
    total += word.blob.size;
  }
  if (total > VOICE_LIMITS.packBytes) return null;
  const origin = raw.origin === 'received' ? 'received' : 'recorded';
  const distributor = isRecord(raw.distributor) && typeof raw.distributor.name === 'string' && raw.distributor.name.trim()
    ? { name: raw.distributor.name.trim().slice(0, VOICE_LIMITS.nameLength) } : undefined;
  const locale = toVoiceLocale(raw.locale);
  const { distributor: _drop, ...rest } = raw;
  return {
    ...(rest as unknown as VoicePackRecord),
    locale,
    vocabularyId: toVoiceLocale(raw.vocabularyId ?? locale),
    origin,
    ...(distributor ? { distributor } : {})
  };
}
```

```ts
// 波形ミニ表示（Ruling 59-2）の縦線の高さ。S8 が録った 16bit PCM の WAV（encodeAudioBufferAsWav の形・ヘッダ 44 バイト）を
// そのまま読む（AudioContext を使わない＝タップの外でも描ける）。形が違えば空（波形を出さない）。値は 0〜1（最大の線が 1）。
export const WAVE_BARS = 40;

export function wavPeaks(bytes: ArrayBuffer, bars = WAVE_BARS): number[] {
  if (bytes.byteLength <= 44) return [];
  const view = new DataView(bytes);
  if (ascii(view, 0) !== 'RIFF' || ascii(view, 8) !== 'WAVE' || ascii(view, 36) !== 'data') return [];
  const channels = view.getUint16(22, true);
  if (channels < 1 || view.getUint16(34, true) !== 16) return [];
  const frames = Math.floor((bytes.byteLength - 44) / (2 * channels));
  if (frames < bars) return [];
  const per = Math.floor(frames / bars);
  const peaks: number[] = [];
  for (let bar = 0; bar < bars; bar += 1) {
    const end = bar === bars - 1 ? frames : (bar + 1) * per;
    let peak = 0;
    for (let frame = bar * per; frame < end; frame += 1) peak = Math.max(peak, Math.abs(view.getInt16(44 + frame * 2 * channels, true)) / 0x8000);
    peaks.push(peak);
  }
  const max = Math.max(...peaks);
  return max > 0 ? peaks.map(peak => Math.round((peak / max) * 100) / 100) : peaks;
}

function ascii(view: DataView, offset: number): string {
  return String.fromCharCode(view.getUint8(offset), view.getUint8(offset + 1), view.getUint8(offset + 2), view.getUint8(offset + 3));
}
```

```ts
// 録れた 1.5 秒が使えるかの判定（Ruling 58-1）。エンジンの検出器は「持続する声の頭」を探す物で、相対しきい値のため
// 声の無い一定ノイズでも先頭で「見つかる」。声が無いことは peakRms の絶対値で別に見る。数値は実機で調整する前提の製品判断。
export const SILENCE_PEAK_RMS = 0.01;   // ≒ −40 dBFS。これ未満は「声が入っていない」
export const LATE_ONSET_MS = 1000;      // 1.5 秒の窓でこれより後に始まった声は切り出し（最長 745ms）が収まらない
export const MIN_TAKE_MS = 600;         // 録れた長さがこれ未満なら失敗（iOS で稀に短い・空の録音）

export type TakeProblem = 'silent' | 'late' | 'short';

export interface AudioBufferLike {
  numberOfChannels: number;
  sampleRate: number;
  length: number;
  duration: number;
  getChannelData(channel: number): Float32Array;
}

export interface TakeCheck { problem: TakeProblem | null; onsetMs: number | null; peakRms: number }

export function checkTake(buffer: AudioBufferLike): TakeCheck {
  if (buffer.duration * 1000 < MIN_TAKE_MS) return { problem: 'short', onsetMs: null, peakRms: 0 };
  let onset: { onsetMs: number; peakRms: number };
  try {
    onset = measureRecordingOnset(buffer as unknown as AudioBuffer);
  } catch {
    return { problem: 'silent', onsetMs: null, peakRms: 0 };
  }
  if (onset.peakRms < SILENCE_PEAK_RMS) return { problem: 'silent', onsetMs: onset.onsetMs, peakRms: onset.peakRms };
  if (onset.onsetMs > LATE_ONSET_MS) return { problem: 'late', onsetMs: onset.onsetMs, peakRms: onset.peakRms };
  return { problem: null, onsetMs: onset.onsetMs, peakRms: onset.peakRms };
}
```

```ts
import { mixAudioBufferToMono } from '../../engine/onset-detector.js';
import { encodeAudioBufferAsWav, normalizeAudioBuffer } from '../../engine/recording-normalizer.js';
import { checkTake, type AudioBufferLike, type TakeProblem } from './take-check.ts';
import { VOICE_LIMITS, type RecordedWord, type VoiceKey } from './store.ts';

// 録れた 1 語を保存する形にする（Ruling 58-3）。全部エンジンの export を呼ぶだけ（エンジンは 0 行変更）:
// decode → 判定 → モノにまとめる → 頭をそろえて切り出す（normalizeAudioBuffer）→ 16bit WAV。サンプルレートは録音のまま。
// 失敗はエンジンの Error を外へ出さず、種類（TakeProblem・'failed'）に変える（画面の文言は messages.ts のキーだけ。Ruling 58-5）。
export interface DecodeContext {
  decodeAudioData(data: ArrayBuffer): Promise<AudioBufferLike>;
  createBuffer(channels: number, length: number, sampleRate: number): AudioBufferLike;
}

export type TakeFailure = TakeProblem | 'failed';
export type TakeResult = { ok: true; word: RecordedWord } | { ok: false; problem: TakeFailure };

export async function processTake(audio: DecodeContext, blob: Blob, key: VoiceKey): Promise<TakeResult> {
  let source: AudioBufferLike;
  try {
    source = await audio.decodeAudioData(await blob.arrayBuffer());
  } catch {
    return { ok: false, problem: 'failed' };
  }
  const check = checkTake(source);
  if (check.problem || check.onsetMs === null) return { ok: false, problem: check.problem ?? 'silent' };
  try {
    const mono = toMono(audio, source);
    const normalized = normalizeAudioBuffer(audio as unknown as AudioContext, mono as unknown as AudioBuffer, key, check.onsetMs);
    const blobOut = encodeAudioBufferAsWav(normalized.buffer);
    if (blobOut.size > VOICE_LIMITS.wordBytes) return { ok: false, problem: 'failed' };
    return {
      ok: true,
      word: { blob: blobOut, mimeType: 'audio/wav', onsetMs: normalized.onsetMs, sourceOnsetMs: check.onsetMs, durationMs: normalized.durationMs, normalized: true, gainDb: 0 }
    };
  } catch {
    return { ok: false, problem: 'failed' };
  }
}

function toMono(audio: DecodeContext, source: AudioBufferLike): AudioBufferLike {
  if (source.numberOfChannels === 1) return source;
  const mono = audio.createBuffer(1, source.length, source.sampleRate);
  mono.getChannelData(0).set(mixAudioBufferToMono(source as unknown as AudioBuffer));
  return mono;
}
```

```ts
import { createVoiceTimingAdjustments } from '../../engine/recorded-voice-pack.js';
import type { AppSettings } from '../settings/settings.ts';
import type { VoicePackRecord } from '../voices/store.ts';
import type { VoiceSelection } from '../voices/selection.ts';

// 声のタイミング（Ruling 62・D24）。「くわしく」の最後の行の「‹ 早く」「遅く ›」。負＝声が早く鳴る（エンジンの userOffsetMs と同じ向き）。
// 保存先: 標準の声は設定の voiceOffsetMsByVoiceId、録音・受け取った声は声そのものの timingAdjustments.globalOffsetMs。
export const VOICE_OFFSET_STEP_MS = 10;
export const VOICE_OFFSET_MAX_MS = 200;

// 保存されていた値を受け取る形にする: 有限の整数で ±200 の内だけ（外れたら null）。
export function normalizeVoiceOffset(value: unknown): number | null {
  if (typeof value !== 'number' || !Number.isInteger(value) || Math.abs(value) > VOICE_OFFSET_MAX_MS) return null;
  return value === 0 ? 0 : value;   // -0 を 0 に
}

export function clampVoiceOffset(value: number): number {
  const rounded = Math.round(value / VOICE_OFFSET_STEP_MS) * VOICE_OFFSET_STEP_MS;
  return Math.max(-VOICE_OFFSET_MAX_MS, Math.min(VOICE_OFFSET_MAX_MS, rounded)) || 0;
}

// 1 回押すと 10ms。端で止まる（端ではボタン自体を押せなくする＝canStep）。
export function stepVoiceOffset(current: number, direction: -1 | 1): number {
  return clampVoiceOffset(clampVoiceOffset(current) + direction * VOICE_OFFSET_STEP_MS);
}

export function canStep(current: number, direction: -1 | 1): boolean {
  return direction < 0 ? current > -VOICE_OFFSET_MAX_MS : current < VOICE_OFFSET_MAX_MS;
}

// 表示（D24）: 0 は「±0 ms」、負は数学のマイナス記号「−20 ms」、正は「+10 ms」。日英共通。
export function formatVoiceOffset(ms: number): string {
  if (ms === 0) return '±0 ms';
  return ms < 0 ? `−${Math.abs(ms)} ms` : `+${ms} ms`;
}

export function voiceOffsetFor(settings: Pick<AppSettings, 'voiceOffsetMsByVoiceId'>, voice: VoiceSelection): number {
  if (voice.kind === 'standard') return settings.voiceOffsetMsByVoiceId[voice.id] ?? 0;
  return voice.pack.timingAdjustments?.globalOffsetMs ?? 0;
}

// 標準の声: 設定への差分（0 なら消す）。
export function chooseStandardVoiceOffset(settings: Pick<AppSettings, 'voiceOffsetMsByVoiceId'>, voiceId: string, value: number): Partial<AppSettings> {
  const { [voiceId]: _old, ...rest } = settings.voiceOffsetMsByVoiceId;
  const next = clampVoiceOffset(value);
  return { voiceOffsetMsByVoiceId: next === 0 ? rest : { ...rest, [voiceId]: next } };
}

// 録音・受け取った声: globalOffsetMs に書いた新しい声（語ごとの値は残す＝旧画面で調整した受け取りの声。Ruling 62-5）。
// 全部 0 になったら timingAdjustments ごと外す。音は変わらないので updatedAt は進めない（decode の記憶を捨てない）。
export function withPackOffset(pack: VoicePackRecord, value: number, now: string): VoicePackRecord {
  const next = clampVoiceOffset(value);
  const words = pack.timingAdjustments?.wordOffsetsMs ?? {};
  const { timingAdjustments: _old, ...rest } = pack;
  if (next === 0 && Object.values(words).every(offset => offset === 0)) return rest;
  return { ...rest, timingAdjustments: createVoiceTimingAdjustments(next, words, now) };
}

// 「‹ 早く」「遅く ›」を 1 回押した結果（純粋）。標準の声は設定の差分、録音・受け取った声は書き換えた声（呼び出し側が保存する）。
export interface VoiceOffsetStep { voice: VoiceSelection; patch: Partial<AppSettings>; pack: VoicePackRecord | null; offsetMs: number }
```

```ts
export class RecorderError extends Error {
  readonly kind: RecorderUnavailable;
  constructor(kind: RecorderUnavailable) {
    super(`recorder ${kind}`);
    this.name = 'RecorderError';

  return MIME_TYPES.find(type => { try { return isSupported(type); } catch { return false; } });

export function recorderErrorKind(error: unknown): RecorderUnavailable {
  if (error instanceof RecorderError) return error.kind;
  const name = typeof error === 'object' && error !== null ? (error as { name?: unknown }).name : undefined;
  return name === 'NotAllowedError' || name === 'SecurityError' ? 'denied' : 'failed';
}

export function requestMicrophone(): Promise<void> {
  if (!recorderSupported()) return Promise.reject(new RecorderError('unsupported'));
  return withRecordingSession(async () => {
    const stream = await openMicrophone();
    stopStream(stream);
  });
}

// 1 語を録る: 開く → 録音開始（onStart）→ ms 後に止める → 閉じる。timeslice は使わない（iOS の MP4 を 1 つの Blob で受ける）。
export function recordTake({ ms = TAKE_MS, onStart }: { ms?: number; onStart?: () => void } = {}): Promise<WordRecording> {
  if (!recorderSupported()) return Promise.reject(new RecorderError('unsupported'));
  return withRecordingSession(async () => {
    const stream = await openMicrophone();
    try {
      return await capture(stream, ms, onStart);
    } finally {
      stopStream(stream);
    }
  });
}

async function openMicrophone(): Promise<MediaStream> {
  try {
    return await navigator.mediaDevices.getUserMedia(CONSTRAINTS);
  } catch (error) {
    throw new RecorderError(recorderErrorKind(error));
  }
}

function stopStream(stream: MediaStream): void {
  for (const track of stream.getTracks()) track.stop();
}

function capture(stream: MediaStream, ms: number, onStart?: () => void): Promise<WordRecording> {
  return new Promise((resolve, reject) => {
    const mimeType = pickMimeType();
    let recorder: MediaRecorder;
    try {
      recorder = new MediaRecorder(stream, mimeType ? { mimeType } : undefined);
    } catch {
      reject(new RecorderError('failed'));
      return;
    }
    const chunks: Blob[] = [];
    let settled = false;
    const finish = (outcome: WordRecording | RecorderError) => {
      if (settled) return;
      settled = true;
      clearTimeout(stopTimer);
      clearTimeout(giveUpTimer);
      if (outcome instanceof RecorderError) reject(outcome);
      else resolve(outcome);
    };
    recorder.ondataavailable = event => { if (event.data.size > 0) chunks.push(event.data); };
    recorder.onerror = () => finish(new RecorderError('failed'));
    recorder.onstop = () => {
      const type = recorder.mimeType || mimeType || chunks[0]?.type || '';
      const blob = new Blob(chunks, type ? { type } : undefined);
      finish(blob.size > 0 ? { blob, mimeType: type } : new RecorderError('failed'));
    };
    try {
      recorder.start();
    } catch {
      finish(new RecorderError('failed'));
      return;
    }
    onStart?.();
    const stopTimer = setTimeout(() => { if (recorder.state !== 'inactive') recorder.stop(); }, ms);
    const giveUpTimer = setTimeout(() => finish(new RecorderError('failed')), ms + STOP_GRACE_MS);
  });
}
```

```ts
export interface PlaybackVoice {
  voice: VoiceSelection | null;
  offsetMs: number;
  // 「‹ 早く」「遅く ›」: 声と差分を返す（設定の差分は画面が change で書く。録音の声はここで端末へ保存する）。

export function usePlaybackVoice(): PlaybackVoice {
  const { settings } = useSettings();
  const storage = useStorage();
  const [voice, setVoice] = useState<VoiceSelection | null>(() => standardSelection(settings.selectedVoiceId));
  const latest = useRef(voice);
  latest.current = voice;
  useEffect(() => {
    let alive = true;
    void resolveVoiceSelection(storage, settings.selectedVoiceId).then(resolved => { if (alive) setVoice(resolved); });
    return () => { alive = false; };
  }, [storage, settings.selectedVoiceId]);
  return {
    voice,
    offsetMs: voice ? voiceOffsetFor(settings, voice) : 0,
    step(current, direction) {
      const now = latest.current;
      if (!now) return null;
      const result = stepVoice(current, now, direction, new Date().toISOString());
      latest.current = result.voice;
      setVoice(result.voice);
      if (result.pack) void saveVoicePack(storage, result.pack);
      return result;
    }
  };
```

```js
const KEYS = ['slow', 'quick', 'one', 'two', 'three', 'four', 'five', 'six', 'seven', 'eight', 'and', 'ah'];

function voice(id, name, locale, recorded, extra = {}) {
  return { schemaVersion: 1, type: 'dance-count-recorded-voice-pack', id, name, locale, vocabularyId: locale, updatedAt: UPDATED, origin: 'recorded', recorded, ...extra };
}

export const UI_CHECK_VOICES = {

// ブラウザの中で動く（page.evaluate に渡す。外の変数は使えない）。recorded 語ぶんの WAV（16bit・モノ・48kHz・0.745 秒・
// 40ms から 0.4 秒の 220Hz＋倍音）を作って voice-packs へ書く。
export async function seedVoicePacksInPage(voices) {
  const keys = ['slow', 'quick', 'one', 'two', 'three', 'four', 'five', 'six', 'seven', 'eight', 'and', 'ah'];
  const rate = 48000;
  const frames = Math.round(rate * 0.745);
  const bytes = new ArrayBuffer(44 + frames * 2);
  const view = new DataView(bytes);
  const text = (offset, value) => { for (let i = 0; i < value.length; i += 1) view.setUint8(offset + i, value.charCodeAt(i)); };
  text(0, 'RIFF'); view.setUint32(4, 36 + frames * 2, true); text(8, 'WAVE'); text(12, 'fmt ');
  view.setUint32(16, 16, true); view.setUint16(20, 1, true); view.setUint16(22, 1, true); view.setUint32(24, rate, true);
  view.setUint32(28, rate * 2, true); view.setUint16(32, 2, true); view.setUint16(34, 16, true); text(36, 'data'); view.setUint32(40, frames * 2, true);
  for (let i = 0; i < frames; i += 1) {
    const t = i / rate;
    const on = t >= 0.04 && t < 0.44 ? 0.5 * (Math.sin(2 * Math.PI * 220 * t) + 0.5 * Math.sin(2 * Math.PI * 440 * t)) / 1.5 : 0;
    view.setInt16(44 + i * 2, Math.round(on * 0x7fff), true);
  }
  const db = await new Promise((resolve, reject) => {
    const request = indexedDB.open('dance-count-v2', 2);
    request.onsuccess = () => resolve(request.result);
    request.onerror = () => reject(request.error);
  });
  await new Promise((resolve, reject) => {
    const transaction = db.transaction('voice-packs', 'readwrite');
    for (const { recorded, ...pack } of voices) {
      const words = {};
      for (const key of keys.slice(0, recorded)) {
        words[key] = { blob: new Blob([bytes], { type: 'audio/wav' }), mimeType: 'audio/wav', onsetMs: 40, sourceOnsetMs: 320, durationMs: 745, normalized: true, gainDb: 0 };
      }
      transaction.objectStore('voice-packs').put({ ...pack, words });
    }
    transaction.oncomplete = resolve;
    transaction.onerror = () => reject(transaction.error);
  });
  db.close();
}

export const UI_CHECK_VOICE_KEYS = KEYS;
```

```js
#!/usr/bin/env node
// 偽マイク用の WAV を作る（Ruling 63-5）。Chrome の --use-file-for-fake-audio-capture に渡す（ループで何度も鳴る）。
// fake-mic.wav: 1.5 秒・48kHz・16bit・モノ。0.3 秒の無音 → 0.6 秒の 220Hz＋倍音（声の代わり）→ 0.6 秒の無音。
// silence.wav : 1.5 秒の全 0。
// 使い方: node tools/make-fake-mic-wav.js（test/fixtures/voice/ に書く。中身は決定的なので何度作っても同じバイト列）
import fs from 'node:fs';
import path from 'node:path';

export const SAMPLE_RATE = 48000;
export const TOTAL_SECONDS = 1.5;
export const VOICE_START_SECONDS = 0.3;
export const VOICE_SECONDS = 0.6;
export const VOICE_PEAK = 0.5;

export function fakeVoiceSamples({ silent = false } = {}) {
  const samples = new Float32Array(Math.round(SAMPLE_RATE * TOTAL_SECONDS));
  if (silent) return samples;
  const start = Math.round(SAMPLE_RATE * VOICE_START_SECONDS);
  const length = Math.round(SAMPLE_RATE * VOICE_SECONDS);
  const ramp = Math.round(SAMPLE_RATE * 0.01);
  for (let i = 0; i < length; i += 1) {
    const t = i / SAMPLE_RATE;
    const envelope = Math.min(1, i / ramp, (length - i) / ramp);
    const tone = Math.sin(2 * Math.PI * 220 * t) + 0.5 * Math.sin(2 * Math.PI * 440 * t) + 0.25 * Math.sin(2 * Math.PI * 660 * t);
    samples[start + i] = VOICE_PEAK * envelope * tone / 1.75;
  }
  return samples;
}

export function encodeMonoWav(samples, sampleRate = SAMPLE_RATE) {
  const buffer = Buffer.alloc(44 + samples.length * 2);
  buffer.write('RIFF', 0, 'ascii');
  buffer.writeUInt32LE(36 + samples.length * 2, 4);
  buffer.write('WAVE', 8, 'ascii');
  buffer.write('fmt ', 12, 'ascii');
  buffer.writeUInt32LE(16, 16);
  buffer.writeUInt16LE(1, 20);
  buffer.writeUInt16LE(1, 22);
  buffer.writeUInt32LE(sampleRate, 24);
  buffer.writeUInt32LE(sampleRate * 2, 28);
  buffer.writeUInt16LE(2, 32);
  buffer.writeUInt16LE(16, 34);
  buffer.write('data', 36, 'ascii');
  buffer.writeUInt32LE(samples.length * 2, 40);
  samples.forEach((value, index) => {
    const clamped = Math.max(-1, Math.min(1, value));
    buffer.writeInt16LE(Math.round(clamped < 0 ? clamped * 0x8000 : clamped * 0x7fff), 44 + index * 2);
  });
  return buffer;
}
```
