**計画者の発見（仕様との差）:**
1. 仕様 §10・Ruling 20 は「`audio-engine.js` と `timeline-view-model.js` の tick→秒を集約」と書くが、**`timeline-view-model.js` に tick→秒の計算は無い**（tick だけで完結）。触らない。代わりに実在する残り2か所 `rhythm-compiler.js` の `eventTimeSeconds` と `dance-timing-audit.js` の ms 変換を集約対象に入れる。
2. `timing:audit` はエンジンの再生処理（`audio-engine.js`）を通らない。**`timing:audit` の一致だけでは集約の無変化を証明できない**ので、Task 1 で再生エンジンの予約時刻を偽の AudioContext で全部記録したゴールデン（`test/engine/fixtures/engine-schedule-golden.json`）を移設前に固定する。
3. 浮動小数の演算順で結果が変わる（実測: `t/960*60/bpm` と `(t/960*60000/bpm)/1000` は 428,580 通り中 72,080 通りで最下位ビットが違う）。tempo-map の constant は既存の式と**演算順まで同じ**にし、秒用 `tickToSeconds`・`secondsToTick` も持たせる（仕様の `tickToMs`・`msToTick` に加えて）。
4. `rhythm-compiler.js` の `eventTimeSeconds` は `(t/960)*(60/bpm)` で、エンジン本体の `t/960*60/bpm` と演算順が違う。集約で `eventTimeSeconds` と `eventOccurrenceTime` だけは最下位ビットが変わりうる（どちらも新旧どの画面の再生経路からも呼ばれていない＝`grep` で確認済み。テストの期待値 11.25・2010.25 は変わらない）。
5. `voice-pack-loader.js` は manifest の `revision` を音声 URL に `?v=` として付ける。これは import のキャッシュバスターではなく「差し替えた音声を確実に読み直す」仕組み（AGENTS.md）なので**残す**。段7の Service Worker で要対応（全体地図の持ち越し表）。
6. 旧画面は `../src/engine/x.js?v=…` と `?v=` 付きで読み、エンジン内部は `?v=` 無しで読むため、旧画面では `voice-pack-loader.js` が2インスタンスになる。旧画面のコードは `instanceof VoicePackError` を使っていない（`grep` 確認）ので実害なし。旧画面は保守停止（Ruling 10）なので HTML の `?v=` は触らない。

```js
// 再生エンジンが実際に予約する時刻を、偽の AudioContext で記録する（tempo-map 集約の前後比較用）。
import { PracticeAudioEngine } from '../../src/v2/audio-engine.js';
import { compileRhythm, compileRoutine } from '../../src/v2/rhythm-compiler.js';
import { DANCE_PROFILES } from '../../src/v2/dance-profiles.js';

const VOICE_KEYS = ['slow', 'quick', 'one', 'two', 'three', 'four', 'five', 'six', 'seven', 'eight', 'and', 'ah'];

class FakeParam {
  constructor(log, name) { this.log = log; this.name = name; this.value = 1; }
  setValueAtTime(value, time) { this.log.push([this.name, 'set', value, time]); }
  exponentialRampToValueAtTime(value, time) { this.log.push([this.name, 'exp', value, time]); }
  linearRampToValueAtTime(value, time) { this.log.push([this.name, 'lin', value, time]); }
}

class FakeNode {
  constructor(log, kind) {
    this.log = log;
    this.kind = kind;
    this.gain = new FakeParam(log, `${kind}.gain`);
    this.frequency = new FakeParam(log, `${kind}.frequency`);
  }
  connect(node) { return node; }
  start(time, offset = 0) { this.log.push([this.kind, 'start', time, offset, this.frequency.value]); }
  stop(time) { this.log.push([this.kind, 'stop', time]); }
}

export class FakeAudioContext {
  constructor() { this.currentTime = 0; this.log = []; this.destination = {}; }
  createOscillator() { return new FakeNode(this.log, 'osc'); }
  createGain() { return new FakeNode(this.log, 'gain'); }
  createBufferSource() { return new FakeNode(this.log, 'voice'); }
}

function fakeVoicePack() {
  return { words: new Map(VOICE_KEYS.map((key, index) => [key, { buffer: { duration: 0.9 }, onsetMs: 37 + index * 3, gainDb: -1.5 }])) };
}

export const SCHEDULE_CASES = [
  ...Object.values(DANCE_PROFILES).flatMap(profile => [profile.practiceBpm, profile.competitionBpm.max, 67].map(bpm => ({
    name: `${profile.id}@${bpm}`,
    compiled: () => compileRhythm({ danceId: profile.id, counts: profile.templates.at(-1) }),
    options: { bpm, countInBars: 1 }
  }))),
  {
    name: 'jive-routine-range@133.7',
    compiled: () => compileRoutine({ danceId: 'jive', figures: [
      { id: 'a', name: 'a', counts: '123a45a6' },
      { id: 'b', name: 'b', counts: '12' },
      { id: 'c', name: 'c', counts: '3a45a6' }
    ] }),
    options: { bpm: 133.7, countInBars: 2, rangeStartTick: 6 * 960, rangeEndTick: 12 * 960, voiceOffsetMs: 23, wordOffsetsMs: { six: -11 } }
  },
  {
    name: 'waltz-no-loop@84',
    compiled: () => compileRhythm({ danceId: 'waltz', counts: '12&3' }),
    options: { bpm: 84, loop: false, countInBars: 0, metronome: false }
  }
];

export async function recordSchedule(testCase) {
  const savedRaf = globalThis.requestAnimationFrame;
  const savedCancel = globalThis.cancelAnimationFrame;
  globalThis.requestAnimationFrame = () => 1;
  globalThis.cancelAnimationFrame = () => {};
  const context = new FakeAudioContext();
  const frames = [];
  const engine = new PracticeAudioEngine(context, { voicePack: fakeVoicePack(), onFrame: frame => frames.push(frame) });
  try {
    const compiled = testCase.compiled();
    const patternStart = engine.start({ compiled, ...testCase.options });
    for (let step = 1; step <= 40; step += 1) {
      context.currentTime = step * 0.173;
      engine.scheduleAhead();
      engine.frame();
    }
    return {
      patternStart,
      log: context.log,
      frames: frames.map(frame => ({
        playing: frame.playing,
        token: frame.event?.token ?? null,
        tick: frame.event?.absoluteTick ?? null,
        progress: frame.progress,
        originalTick: frame.originalTick ?? null,
        iteration: frame.iteration ?? null,
        countIn: frame.countIn ?? null
      }))
    };
  } finally {
    engine.stop();
    globalThis.requestAnimationFrame = savedRaf;
    globalThis.cancelAnimationFrame = savedCancel;
  }
}
```

`src/engine/tempo-map.js`:
```js
// テンポ写像（Ruling 20）: 整数 tick ↔ 時刻 の変換はこのファイルだけで行う。
// constant の式は移設前の audio-engine.js / dance-timing-audit.js と演算順まで同じにしてある
// （浮動小数の丸めまで一致させ、tempo-map 集約で鳴る時刻が1ビットも変わらないようにするため）。
import { TICKS_PER_BEAT } from './dance-profiles.js';

/**
 * 曲の口（Ruling 20・型だけ。v1 は実データなし）。
 * @typedef {{ kind: 'constant', bpm: number, firstBeatMs: number } | { kind: 'beatMap', beatsMs: number[] }} TrackTempo
 * @typedef {{ kind: 'CC0' | 'PD-recording' | 'CC-BY', source: string, url: string, note: string }} TrackLicense
 * @typedef {{ id: string, danceId: string, title: string, file: string, durationMs: number, beatsPerBar: number,
 *   tempo: TrackTempo, license: TrackLicense, measuredAt: string }} TrackProfile
 */

/**
 * @typedef {object} TempoMap
 * @property {'constant' | 'beatMap'} kind
 * @property {(tick: number) => number} tickToMs
 * @property {(ms: number) => number} msToTick
 * @property {(tick: number) => number} tickToSeconds
 * @property {(seconds: number) => number} secondsToTick
 */

/** @returns {TempoMap & { kind: 'constant', bpm: number }} */
export function constantTempo(bpm) {
  if (!(bpm > 0) || !Number.isFinite(Number(bpm))) throw new RangeError('BPM must be greater than zero');
  return Object.freeze({
    kind: 'constant',
    bpm,
    tickToMs: tick => tick / TICKS_PER_BEAT * 60_000 / bpm,
    msToTick: ms => ms / (60_000 / bpm) * TICKS_PER_BEAT,
    tickToSeconds: tick => tick / TICKS_PER_BEAT * 60 / bpm,
    secondsToTick: seconds => seconds / (60 / bpm) * TICKS_PER_BEAT
  });
}

/**
 * 実測した拍の時刻（ms）の間を区分線形で補間する。範囲外は端の拍間隔で外挿する
 * （最後の拍より後は最後の間隔、最初の拍より前＝弱起は最初の間隔）。
 * @returns {TempoMap & { kind: 'beatMap', beatsPerBar: number }}
 */
export function beatMapTempo(beatsMs, beatsPerBar) {
  if (!Array.isArray(beatsMs) || beatsMs.length < 2) throw new RangeError('A beat map needs at least two beats');
  if (!Number.isInteger(beatsPerBar) || beatsPerBar < 1) throw new RangeError('beatsPerBar must be a positive integer');
  for (let index = 0; index < beatsMs.length; index += 1) {
    if (!Number.isFinite(beatsMs[index])) throw new RangeError(`Beat ${index} is not a finite time`);
    if (index > 0 && !(beatsMs[index] > beatsMs[index - 1])) throw new RangeError(`Beat ${index} is not after beat ${index - 1}`);
  }
  const beats = Object.freeze([...beatsMs]);
  const last = beats.length - 1;

  const tickToMs = tick => {
    const beat = tick / TICKS_PER_BEAT;
    const index = Math.min(Math.max(Math.floor(beat), 0), last - 1);
    return beats[index] + (beat - index) * (beats[index + 1] - beats[index]);
  };
  const msToTick = ms => {
    let index = 0;
    if (ms >= beats[last]) index = last - 1;
    else if (ms > beats[0]) while (beats[index + 1] <= ms) index += 1;
    return (index + (ms - beats[index]) / (beats[index + 1] - beats[index])) * TICKS_PER_BEAT;
  };

  return Object.freeze({
    kind: 'beatMap',
    beatsPerBar,
    tickToMs,
    msToTick,
    tickToSeconds: tick => tickToMs(tick) / 1000,
    secondsToTick: seconds => msToTick(seconds * 1000)
  });
}
```

注: 仕様は「範囲外は最後の拍間隔で外挿」とだけ書く。最初の拍より前（弱起・`leadingSubdivision:'pickup'` で tick が負になる）は仕様が触れていないため、**最初の間隔**で外挿する解釈を採った（拍間隔が最も近い区間を使う）。段2以降で曲を載せる時（第2弾）に Fable の確認を取ること。

`constantTempo` は文字列の `'84'` を拒否しない（移設前の `!(bpm > 0)` と同じ判定。旧画面は `Number(...)` で渡しているが、互換のため厳しくしない）。

```ts
const JAPANESE_FORBIDDEN = [
  'BPM', 'インポート', 'エクスポート', '読み込み', '書き出し', 'カウントイン', '音声パック', 'ボイスパック',
  '設定を保存', 'ストレージ', 'キャッシュ', '同期', 'エラーコード'
] as const;

export function findForbiddenWords(text: string): string[] {
  const found: string[] = JAPANESE_FORBIDDEN.filter(word => text.includes(word));

  return found;
```

```ts
// ハッシュ ↔ 画面の対応表（仕様 §3）。純粋関数だけ。window には触らない（platform/hash.ts の役目）。
export type ScreenId = 'S1' | 'S2' | 'S3' | 'S4' | 'S5' | 'S6' | 'S7' | 'S8' | 'S9' | 'S10' | 'S11' | 'S12' | 'S13';
export type ScreenKind = 'root' | 'focus' | 'list';
export type TabId = 'practice' | 'routines' | 'settings';
export type RouteParams = { danceId?: string; routineId?: string };
export interface Route { screen: ScreenId; params: RouteParams }

interface ScreenSpec { kind: ScreenKind; parent: ScreenId | null; tab: TabId; path: string }

// path の ":name" は ID 1区切り（英数字・ハイフン・下線のみ）。
export const SCREENS: Record<ScreenId, ScreenSpec> = {
  S1: { kind: 'root', parent: null, tab: 'practice', path: '/' },
  S2: { kind: 'focus', parent: 'S1', tab: 'practice', path: '/practice/:danceId' },
  S3: { kind: 'root', parent: null, tab: 'routines', path: '/routines' },
  S4: { kind: 'focus', parent: 'S3', tab: 'routines', path: '/routines/new' },
  S5: { kind: 'focus', parent: 'S3', tab: 'routines', path: '/routines/:routineId/play' },
  S6: { kind: 'root', parent: null, tab: 'settings', path: '/settings' },
  S7: { kind: 'list', parent: 'S6', tab: 'settings', path: '/voices' },
  S8: { kind: 'focus', parent: 'S7', tab: 'settings', path: '/voices/record' },
  S9: { kind: 'focus', parent: 'S7', tab: 'settings', path: '/share' },
  S10: { kind: 'focus', parent: 'S7', tab: 'settings', path: '/receive' },
  S11: { kind: 'focus', parent: 'S6', tab: 'settings', path: '/help/sound' },
  S12: { kind: 'list', parent: 'S6', tab: 'settings', path: '/help/install' },
  S13: { kind: 'list', parent: 'S6', tab: 'settings', path: '/about' }
};

// S4 は「新規」と「編集」の2つの形を持つので、編集の形をここで足す。
const EXTRA_PATHS: Array<{ screen: ScreenId; path: string }> = [{ screen: 'S4', path: '/routines/:routineId/edit' }];

const ID_SEGMENT = /^[A-Za-z0-9_-]+$/;
const HOME: Route = { screen: 'S1', params: {} };

const PATTERNS = [
  ...(Object.entries(SCREENS) as Array<[ScreenId, ScreenSpec]>).map(([screen, spec]) => ({ screen, path: spec.path })),
  ...EXTRA_PATHS
].map(({ screen, path }) => ({ screen, segments: path.split('/').filter(Boolean) }));

export function parseHash(hash: string): Route {
  const path = hash.replace(/^#/, '').split('?')[0] ?? '';
  const segments = path.split('/').filter(Boolean);
  for (const pattern of PATTERNS) {
    if (pattern.segments.length !== segments.length) continue;
    const params: Record<string, string> = {};
    const matched = pattern.segments.every((expected, index) => {
      const actual = segments[index] ?? '';
      if (expected.startsWith(':')) {
        if (!ID_SEGMENT.test(actual)) return false;
        params[expected.slice(1)] = actual;
        return true;
      }
      return expected === actual;
    });
    if (matched) return { screen: pattern.screen, params };
  }
  return { ...HOME, params: {} };
}

export function buildHash(route: Route): string {
  const params = route.params as Record<string, string | undefined>;
  const candidates = PATTERNS.filter(pattern => pattern.screen === route.screen);
  for (const pattern of candidates) {
    const names = pattern.segments.filter(segment => segment.startsWith(':')).map(segment => segment.slice(1));
    const given = Object.keys(params).filter(name => params[name] !== undefined);
    if (names.length !== given.length || !names.every(name => given.includes(name))) continue;
    const path = pattern.segments.map(segment => (segment.startsWith(':') ? params[segment.slice(1)] : segment)).join('/');
    return `#/${path}`;
  }
  throw new Error(`No hash for ${route.screen} with params ${JSON.stringify(route.params)}`);
}

export function showsTabs(screen: ScreenId): boolean {
  return SCREENS[screen].kind !== 'focus';
}

export function parentScreen(screen: ScreenId): ScreenId | null {
  return SCREENS[screen].parent;
}

export function tabOf(screen: ScreenId): TabId {
  return SCREENS[screen].tab;
}
```

```ts
export function onHashChange(listener: (hash: string) => void): () => void {
  const handler = () => listener(window.location.hash);
  window.addEventListener('hashchange', handler);
  return () => window.removeEventListener('hashchange', handler);
}

// replace=true は履歴を積まない（戻る先が無い時の親への置き換え。仕様 §3）。
export function setHash(hash: string, { replace = false }: { replace?: boolean } = {}): void {
  if (replace) {
    window.history.replaceState(null, '', hash);
    window.dispatchEvent(new HashChangeEvent('hashchange'));
  } else {

  }
}
```

```tsx
import type { ComponentType } from 'preact';
import { useMemo, useState } from 'preact/hooks';
import { TabBar } from './components/TabBar.tsx';
import { createAppTranslator, resolveAppLocale, type AppLocale } from './i18n/index.ts';
import { deviceLanguages } from './platform/locale.ts';
import { showsTabs, tabOf, type ScreenId } from './router/routes.ts';
import { HomeScreen } from './screens/HomeScreen.tsx';
import { RoutinesScreen } from './screens/RoutinesScreen.tsx';
import { SettingsScreen } from './screens/SettingsScreen.tsx';
import { I18nContext } from './state/i18n.ts';
import { useRoute } from './state/route.ts';

// 段1で中身があるのは S1・S3・S6 だけ。未実装の画面はホームを出す（段2以降でここに足す）。
const SCREEN_COMPONENTS: Partial<Record<ScreenId, ComponentType>> = {
  S1: HomeScreen,
  S3: RoutinesScreen,
  S6: SettingsScreen
};

export function App() {
  const route = useRoute();
  const [locale, setLocale] = useState<AppLocale>(() => resolveAppLocale(deviceLanguages()));
  const i18n = useMemo(() => ({ translator: createAppTranslator(locale), setLocale }), [locale]);
  const screen: ScreenId = SCREEN_COMPONENTS[route.screen] ? route.screen : 'S1';
  const Screen = SCREEN_COMPONENTS[screen] ?? HomeScreen;
  return (
    <I18nContext.Provider value={i18n}>
      <div class="dc-shell" lang={locale}>
        <Screen />
        {showsTabs(screen) && <TabBar current={tabOf(screen)} />}
      </div>
    </I18nContext.Provider>
  );
}
```
