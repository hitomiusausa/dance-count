# Dance Count アプリ化 v1 — 段5「配る・受け取る（S9・S10）」 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 先生が録音した声とルーティンを 1 つのファイル（`.dancecount.txt`）にして共有シート／ダウンロードで配り（S9）、生徒は LINE 等で受け取ったファイルを「ファイルを選ぶ → 使う → ホームへ」の 3 タップで取り込み、そのままワルツが先生の声で鳴る（S10）。壊れた・大きすぎ・新しすぎのファイルは選ぶ画面の枠で止まる。

**Tech Stack:** Node ≥22.18（検証は v24.13.0）、Vite 8.3.4、Preact 10.29.8、TypeScript 6.0.3、node:test、playwright-core 1.61.1（既存）。Web API: `navigator.share`/`canShare`（Web Share Level 2）・`<a download>`・`navigator.clipboard.writeText`・`File.text()`・Cache API（受け口だけ）。**新しい依存なし**。

## 計画者の発見（裁定の範囲で解決済み。台帳に記録する）

1. **声の「同じ先生」は 配る人の名前＋先生の端末の ID＋言葉 で見分ける**: 録音の声の ID は全員 `user-recorded-v1`／`user-recorded-ja-v1` の固定（0-1）なので、`sourceId`＋`locale` だけだと**山田先生と佐藤先生の英語の声が互いに置き換わる**。Ruling 84-4 の括弧（「旧単体パックは distributor が無いので sourceId＋locale」）と Ruling 83 の代償（「山田（大会用）にすれば別の声になる」）から、束では `distributor.name` も鍵に入れる（旧単体パックは名前なし同士）。→ Open question 1。
2. **受け取った声は `normalizeVoicePackRecord` を通してから計画に入れる**: 端末に書いた後の再読み込みで外れる物（名前 41 字・語ごと 2MB 超など）を、確認の画面に出す前に「読めなかった」にする（書いたのに一覧に出ない、を作らない）。
3. **`sourceUpdatedAt` は import の前に読む**（実証 3）。
4. **同意の文をボタンの中に入れると U10（ボタンの文言 2 行まで）で落ちる**（375 で 3 行）。→ 四角だけを `button role="checkbox"`（48×48・`aria-labelledby` で文を指す）、文はその横の `div`（押しても切り替わる）。→ Open question 4。
5. **S8 完成の 1 行は `<p>` にすると U1（本文 18px）で落ちる** → `div.dc-hint`（16px）。
6. **S7 の「ファイルから受け取る」は一覧の最後の行と 1px しか離れず U3 で落ちる** → 上に 24px・「自分の声を録音する」との間 12px。
7. **S3 の開いた行の「編集」「配る」「削除」は 320 幅で「編／集」のように語の途中で折れる**（ui:check は 1 つの文字列 2 行までを許すので落ちない）→ `white-space: nowrap`・入らない幅では 2 段に並ぶ（段4の S7 と同じ型）。

export interface BundleInput {
  distributorName: string;
  voice: ExportedVoice | null;       // exportVoiceForBundle の結果（声を入れない時は null）
  routines: readonly Routine[];
  now: string;

export async function exportVoiceForBundle(pack: VoicePackRecord): Promise<ExportedVoice> {
  if (pack.origin !== 'recorded' || !isCompletePack(pack)) throw new Error('only a complete recorded voice can be shared');
  return JSON.parse(await exportRecordedVoicePack(pack)) as ExportedVoice;
}

export function cleanDistributorName(name: string): string {
  return [...name.trim()].slice(0, DISTRIBUTOR_NAME_MAX).join('');
}

// 束の JSON 文字列。名前が空・何も入っていない・上限を超える時は投げる（S9 は主ボタンを押せなくしているので、ここに来ない）。
export function buildBundle({ distributorName, voice, routines, now }: BundleInput): string {
  const name = cleanDistributorName(distributorName);
  if (!name) throw new Error('distributor name is required');
  if (!voice && routines.length === 0) throw new Error('nothing to share');
  if (routines.length > BUNDLE_ROUTINES_MAX) throw new Error('too many routines');
  const bundle: Record<string, unknown> = {
    schemaVersion: BUNDLE_SCHEMA_VERSION,
    type: BUNDLE_TYPE,
    createdAt: now,
    createdWith: { app: 'dance-count', version: APP_VERSION },
    distributor: { name }
  };
  if (voice) {
    bundle.consent = { voiceDistribution: true, consentedAt: now } satisfies Consent;
    bundle.voicePack = voice;
  }
  // ルーティンは自分で作った物の形で書く（受け取った物を配り直す時も。元の先生の名前は残さない。Ruling 80-2）。
  if (routines.length > 0) bundle.routines = routines.map(toSharedRoutine);
  return JSON.stringify(bundle);
}

function toSharedRoutine(routine: Routine): Routine {
  const { receivedFrom: _from, ...rest } = routine;
  return { ...rest, source: 'user-created', provenance: { ...USER_PROVENANCE } };
}

export type BundleProblem = 'broken' | 'too-new';

export interface ParsedVoice {
  pack: VoicePackRecord;                 // 先生の端末の ID のまま（受け取る時に付け直す。Ruling 84-1）
  sourceUpdatedAt: string | null;        // 先生の声の更新時刻（importRecordedVoicePack は updatedAt を今に変えるので、その前に読む）
}

export type ParsedBundle =
  | {
      ok: true;
      legacy: boolean;                   // 旧画面の単体パック（dance-count-recorded-voice-pack）
      distributor: { name: string } | null;   // 旧単体パックだけ null
      consent: Consent | null;
      voice: ParsedVoice | null;
      routines: Routine[];               // 鳴らせる物だけ（ID は先生の端末のまま）
      rejected: Array<{ name: string | null }>;   // 読めなかったルーティン（名前が読めなければ null。画面が「（名前なし）」と出す）
    }
  | { ok: false; problem: BundleProblem };

const BROKEN = { ok: false, problem: 'broken' } as const;

// 読む順（Ruling 80-5）: JSON → 種類と版 → 声 → ルーティン → 空でないか。'too-large' は読む前（platform/files.ts）。
export function parseBundle(text: string): ParsedBundle {
  let data: unknown;
  try { data = JSON.parse(text); } catch { return BROKEN; }
  if (!isRecord(data)) return BROKEN;

  let legacy = false;
  let rawVoice: unknown = null;
  let rawRoutines: unknown = undefined;
  let distributor: { name: string } | null = null;
  let consent: Consent | null = null;
  if (data.type === BUNDLE_TYPE) {
    if (Number.isInteger(data.schemaVersion) && (data.schemaVersion as number) > BUNDLE_SCHEMA_VERSION) return { ok: false, problem: 'too-new' };
    if (data.schemaVersion !== BUNDLE_SCHEMA_VERSION) return BROKEN;
    const name = isRecord(data.distributor) && typeof data.distributor.name === 'string' ? cleanDistributorName(data.distributor.name) : '';
    if (!name) return BROKEN;
    distributor = { name };
    rawVoice = data.voicePack ?? null;
    rawRoutines = data.routines;
    if (rawVoice !== null) {
      if (!isRecord(data.consent) || data.consent.voiceDistribution !== true) return BROKEN;
      consent = { voiceDistribution: true, consentedAt: typeof data.consent.consentedAt === 'string' ? data.consent.consentedAt : '' };
    }
  } else if (data.type === LEGACY_VOICE_TYPE) {
    legacy = true;
    rawVoice = data;
  } else {
    return BROKEN;
  }

  let voice: ParsedVoice | null = null;
  if (rawVoice !== null) {
    voice = readVoice(rawVoice);
    if (!voice) return BROKEN;
  }

  const routines: Routine[] = [];
  const rejected: Array<{ name: string | null }> = [];
  if (rawRoutines !== undefined) {
    if (!Array.isArray(rawRoutines) || rawRoutines.length > BUNDLE_ROUTINES_MAX) return BROKEN;
    for (const raw of rawRoutines) {
      const routine = normalizeRoutine(raw);
      if (routine && routinePlayability(routine) === 'ok') routines.push(routine);
      else rejected.push({ name: readableName(raw) });
    }
  }
  if (!voice && routines.length === 0) return BROKEN;   // 空の束・全部読めない束は「読めなかった」と同じ（Ruling 80-5-5）
  return { ok: true, legacy, distributor, consent, voice, routines, rejected };
}

// 声はエンジンの import（base64 → Blob・形の検査）→ 12 語そろい → 端末に保存できる形（名前・語ごと 2MB・合計 20MB）。
function readVoice(raw: unknown): ParsedVoice | null {
  if (!isRecord(raw)) return null;
  let imported: Record<string, unknown>;
  try { imported = importRecordedVoicePack(JSON.stringify(raw)) as Record<string, unknown>; } catch { return null; }
  const pack = normalizeVoicePackRecord({ ...imported, origin: 'recorded' });
  if (!pack || !isCompletePack(pack)) return null;
  return { pack, sourceUpdatedAt: typeof raw.updatedAt === 'string' ? raw.updatedAt : null };
}

function readableName(raw: unknown): string | null {
  if (!isRecord(raw) || typeof raw.name !== 'string' || !raw.name.trim()) return null;
  return [...raw.name.trim()].slice(0, ROUTINE_NAME_MAX).join('');
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

// 配布ファイルの名前（Ruling 80-3・D36）: 声-<先生名>-<YYYY-MM-DD>.dancecount.txt（声が無ければ ルーティン-…）。英語の画面は voice-／routines-。
// ファイル名に使えない文字（/ \ : * ? " < > | と制御文字）は _。名前が空なら 名前なし／unnamed。日付は端末の時計の日付。
const UNSAFE = /[/\\:*?"<>|\u0000-\u001f\u007f]/g;

export function bundleFileName({ locale, distributorName, hasVoice, now }: { locale: AppLocale; distributorName: string; hasVoice: boolean; now: Date }): string {
  const ja = locale === 'ja-JP';
  const prefix = hasVoice ? (ja ? '声' : 'voice') : (ja ? 'ルーティン' : 'routines');
  const name = distributorName.trim().replace(UNSAFE, '_') || (ja ? '名前なし' : 'unnamed');

}

function localDate(now: Date): string {
  const pad = (value: number) => String(value).padStart(2, '0');
  return `${now.getFullYear()}-${pad(now.getMonth() + 1)}-${pad(now.getDate())}`;
}

import { ROUTINES_MAX } from '../routines/limits.ts';
import { USER_PROVENANCE, type Routine } from '../routines/routine.ts';
import { VOICE_LIMITS, type VoicePackRecord } from '../voices/store.ts';
import type { ParsedBundle } from './bundle.ts';

// 受け取る前の計画（S10 の「確認」の中身。Ruling 83-2・84。純粋）: ID を付け直し、同じ物は置き換え、上限で止める。
type ParsedOk = Extract<ParsedBundle, { ok: true }>;

export interface ImportPlan {
  voice: { pack: VoicePackRecord; replaces: VoicePackRecord | null } | null;
  voiceBlocked: boolean;           // 声が 20 に当たる（置き換えでない追加）＝このファイルは受け取らない（選ぶ画面の枠）
  routines: Array<{ routine: Routine; replaces: string | null }>;
  routinesOverLimit: number;       // 200 本に入らなかった本数
  rejected: Array<{ name: string | null }>;
  distributor: { name: string } | null;
}

export interface PlanContext {
  now: string;
  newId(prefix: 'v' | 'r' | 'f'): string;
}

export function planImport(parsed: ParsedOk, packs: readonly VoicePackRecord[], routines: readonly Routine[], { now, newId }: PlanContext): ImportPlan {
  let voice: ImportPlan['voice'] = null;
  let voiceBlocked = false;
  if (parsed.voice) {
    const source = parsed.voice.pack;
    // 同じ先生の同じ言葉の声＝受け取った声のうち、配る人の名前・先生の端末の ID・言葉が同じ物（旧単体パックは名前なし同士。Ruling 84-4）。
    const same = packs.find(pack => pack.origin === 'received' && pack.sourceId === source.id && pack.locale === source.locale
      && (pack.distributor?.name ?? null) === (parsed.distributor?.name ?? null)) ?? null;
    if (!same && packs.length >= VOICE_LIMITS.packs) voiceBlocked = true;
    else voice = { pack: receivedPack(parsed, now, newId('v')), replaces: same };
  }

  const planned: ImportPlan['routines'] = [];
  let room = ROUTINES_MAX - routines.length;
  let routinesOverLimit = 0;
  for (const routine of parsed.routines) {
    const same = routines.find(item => item.receivedFrom?.sourceId === routine.id) ?? null;
    if (!same && room <= 0) { routinesOverLimit += 1; continue; }
    if (!same) room -= 1;
    planned.push({ routine: receivedRoutine(routine, parsed.distributor?.name ?? '', now, newId), replaces: same?.id ?? null });
  }
  return { voice, voiceBlocked, routines: planned, routinesOverLimit, rejected: parsed.rejected, distributor: parsed.distributor };
}

// 受け取った声（Ruling 84-1）: 新しい v- の ID・origin:'received'・誰から・同意・先生の端末の ID と時刻。
// 声のタイミングは先生の端末の遅れ（globalOffsetMs）を捨て、語ごとの癖（wordOffsetsMs）だけ残す。両方 0 なら持たない。
function receivedPack(parsed: ParsedOk, now: string, id: string): VoicePackRecord {
  const { timingAdjustments, distributor: _d, consent: _c, sourceId: _s, sourceUpdatedAt: _u, receivedAt: _r, ...rest } = parsed.voice!.pack;
  const wordOffsetsMs = timingAdjustments?.wordOffsetsMs ?? {};
  const keepWords = Object.values(wordOffsetsMs).some(value => value !== 0);
  return {
    ...rest,
    id,
    origin: 'received',
    updatedAt: now,
    ...(parsed.distributor ? { distributor: parsed.distributor } : {}),
    ...(parsed.consent ? { consent: parsed.consent } : {}),
    ...(keepWords ? { timingAdjustments: { globalOffsetMs: 0, wordOffsetsMs, adjustedAt: timingAdjustments!.adjustedAt } } : {}),
    sourceId: parsed.voice!.pack.id,
    ...(parsed.voice!.sourceUpdatedAt ? { sourceUpdatedAt: parsed.voice!.sourceUpdatedAt } : {}),
    receivedAt: now
  };
}

// 受け取ったルーティン（Ruling 84-2）: ルーティンもフィガーも新しい ID・source:'received'・誰から（先生の端末の ID を sourceId に）。
function receivedRoutine(routine: Routine, distributor: string, now: string, newId: PlanContext['newId']): Routine {
  return {
    ...routine,
    id: newId('r'),
    figures: routine.figures.map(figure => ({ ...figure, id: newId('f') })),
    source: 'received',
    provenance: { ...USER_PROVENANCE, sourceKind: 'received' },
    receivedFrom: { distributor, receivedAt: now, sourceId: routine.id },
    createdAt: now,
    updatedAt: now
  };
}

export interface ImportWriter {
  saveVoice(pack: VoicePackRecord): Promise<boolean>;
  deleteVoice(id: string): Promise<boolean>;
  saveRoutine(routine: Routine): Promise<boolean>;
  removeRoutine(id: string): Promise<boolean>;
}

// 「使う」: 新しい物を先に書き、書けてから古い物を消す（途中で失敗しても、前から持っていた物は消えない）。
// 1 つでも書けなければ false（S10 は確認の画面に留まり「保存できませんでした」。Ruling 83-3）。
export async function commitImport(plan: ImportPlan, writer: ImportWriter): Promise<boolean> {
  if (plan.voice) {
    if (!(await writer.saveVoice(plan.voice.pack))) return false;
    if (plan.voice.replaces) await writer.deleteVoice(plan.voice.replaces.id);
  }
  for (const { routine, replaces } of plan.routines) {
    if (!(await writer.saveRoutine(routine))) return false;
    if (replaces) await writer.removeRoutine(replaces);
  }
  return true;
}

import { BUNDLE_MAX_BYTES } from '../share/limits.ts';

// 配る（共有・保存）と受け取る（ファイルを読む・Android の受け口）の端末の面（Ruling 79-2・81）。
// navigator.share・<a download>・clipboard・Cache API はここだけ。ファイルを選ぶ <input type=file> は S10 の DOM が持つ。
export type ShareOutcome = 'shared' | 'cancelled' | 'refused';

export class BundleTooLargeError extends Error {
  constructor() {
    super('bundle file is too large');
    this.name = 'BundleTooLargeError';
  }
}

// canShare は種類を見ない（F20: Chrome は .json でも true を返し、share() で拒む）。拒否は shareFile の 'refused' で受ける。
export function canShareFile(file: File): boolean {
  try {
    return typeof navigator.share === 'function' && navigator.canShare?.({ files: [file] }) === true;
  } catch {
    return false;
  }
}

let inFlight: Promise<ShareOutcome> | null = null;

// タップの処理の中で、最初の文で・await を挟まずに呼ぶ（F22: WebKit は await 1 つで活性を失う）。
// 前の共有が終わる前にもう一度押されたら、新しく呼ばずに前の結果を返す（F22: 二重に呼ぶと NotAllowedError）。
// 取り消し（AbortError）は 'cancelled'、それ以外の拒否（NotAllowedError 等。F19）は 'refused'。
export function shareFile(file: File): Promise<ShareOutcome> {
  if (inFlight) return inFlight;
  let started: Promise<void>;
  try {
    started = navigator.share({ files: [file] });
  } catch (error) {
    return Promise.resolve(shareFailure(error));
  }
  const result = started.then((): ShareOutcome => 'shared', shareFailure).finally(() => { inFlight = null; });
  inFlight = result;
  return result;
}

function shareFailure(error: unknown): ShareOutcome {
  return error instanceof Error && error.name === 'AbortError' ? 'cancelled' : 'refused';
}

const REVOKE_AFTER_MS = 60_000;

// 共有できないブラウザ（PC など）・共有を拒まれた時: <a download> を同期にクリックして保存する（§11「案内なし。ファイル名を表示」）。
export function downloadFile(file: File): void {
  const url = URL.createObjectURL(file);
  const link = document.createElement('a');
  link.href = url;
  link.download = file.name;
  link.rel = 'noopener';
  link.style.display = 'none';
  document.body.append(link);
  link.click();
  link.remove();
  setTimeout(() => URL.revokeObjectURL(url), REVOKE_AFTER_MS);
}

// 案内文のコピー。writeText はタップの中で最初に呼ぶ（Safari）。clipboard が無い・拒まれたら false（S9 は長押しで選べる欄を出す）。
export function copyText(text: string): Promise<boolean> {
  try {
    if (typeof navigator.clipboard?.writeText !== 'function') return Promise.resolve(false);
    return navigator.clipboard.writeText(text).then(() => true, () => false);
  } catch {
    return Promise.resolve(false);
  }
}

// 20MB を超えるファイルは読まずに拒む（§7・Ruling 80-4）。
export async function readBundleText(file: File): Promise<string> {
  if (file.size > BUNDLE_MAX_BYTES) throw new BundleTooLargeError();
  return file.text();
}

// Android の共有（share_target）の受け口の契約（Ruling 79-2）。段7の Service Worker が POST の 'bundle' を
// cache 'dance-count-inbox' の './inbox/bundle' に Response として put し（ヘッダー 'X-File-Name' に encodeURIComponent したファイル名）、
// './#/receive' へ 303 で送る。S10 はマウント時にこれを呼び、あれば「確認」から始める。読んだら消す（2 回受け取らない）。
export const INBOX_CACHE = 'dance-count-inbox';
export const INBOX_KEY = './inbox/bundle';
const INBOX_FALLBACK_NAME = 'bundle.dancecount.txt';

export async function takeInboxFile(): Promise<File | null> {
  if (typeof caches === 'undefined') return null;
  try {
    const cache = await caches.open(INBOX_CACHE);
    const response = await cache.match(INBOX_KEY);
    if (!response) return null;
    const blob = await response.blob();
    await cache.delete(INBOX_KEY);
    const header = response.headers.get('X-File-Name');
    let name = INBOX_FALLBACK_NAME;
    try { if (header) name = decodeURIComponent(header); } catch { /* 名前が読めなくても中身は受け取る */ }
    return new File([blob], name, { type: blob.type });
  } catch {
    return null;
  }
}

export function ShareScreen() {

  const [intent] = useState(() => takeShareIntent());
  const [voices, setVoices] = useState<VoicePackRecord[] | null>(null);

  const [voiceId, setVoiceId] = useState<string | null>(null);
  const [consent, setConsent] = useState(false);
  const [chosen, setChosen] = useState<string[]>(() => (intent?.routineId ? [intent.routineId] : []));
  const [exported, setExported] = useState<{ id: string; voice: ExportedVoice } | null>(null);
  const [result, setResult] = useState<Result | null>(null);
  const [copy, setCopy] = useState<'idle' | 'copied' | 'manual'>('idle');
  const resultRef = useRef<HTMLDivElement>(null);

  // 配れる声＝録音した完成の声（最大 2）。S7・S8 から来た時はその声を選んだ状態で始める（同意はまだ）。
  useEffect(() => {
    let alive = true;
    void listVoicePacks(storage).then(list => {
      if (!alive) return;
      const recorded = list.filter(pack => pack.origin === 'recorded' && isCompletePack(pack));
      setVoices(recorded);
      if (intent?.voiceId && recorded.some(pack => pack.id === intent.voiceId)) setVoiceId(intent.voiceId);

    setName(next);

    setChosen(current => (current.includes(routine.id) ? current.filter(id => id !== routine.id) : [...current, routine.id]));
  };

  if (voices === null) return <main class="dc-screen dc-screen--focus" data-screen="S9" data-loading><BackButton screen="S9" /></main>;

              <span>{translator.t('saveFileAgain')}</span>
            </button>
          )}
          <button type="button" class="dc-button dc-button--secondary dc-wide" onClick={copyInstructions}>
            <Icon name="copy" /><span>{translator.t('copyInstructions')}</span>
          </button>
          {copy === 'manual' && (
            <div class="dc-field">
              <textarea class="dc-input dc-instructions" readOnly rows={8} value={instructions} aria-label={translator.t('instructionsLabel')} />

          </div>
          {needsConsent && <div class="dc-hint">{translator.t('shareConsentNeeded')}</div>}

                      <b>{routine.name}</b>
                      <small>{count === 1 ? translator.t('routineSummaryOne', { dance }) : translator.t('routineSummary', { dance, count })}</small>
                    </span>
                  </button>
                </div>
              );
            })}
          </div>
        )}
      <div class="dc-dock dc-share-dock">
        {needsSomething && <div class="dc-dock-note">{translator.t('shareChooseSomething')}</div>}
        <button type="button" class="dc-button dc-primary dc-dock-button" disabled={!file} onClick={create}>
          <Icon name="share" /><span>{translator.t('createFile')}</span>
        </button>
      </div>
    </main>
  );
}

// 枠（[data-alert]）は選ぶ画面に 1 つだけ（読めない・大きすぎ・新しすぎ・声が 20・LINE の中）。確認の画面の枠は「保存できませんでした」だけ。
// 同じ先生の声・同じルーティンは「使う」で置き換える（D38）。受け取った声はその場で選ばれる（Ruling 83-3）。
// Android の共有（段7の Service Worker）から来た時は受け口（inbox）のファイルで「確認」から始まる（Ruling 79-2）。
type Step = 'pick' | 'confirm' | 'done';
type PickAlert = { kind: 'broken'; file: string } | { kind: 'too-large' | 'too-new' | 'voice-limit' } | null;
const NAMES_SHOWN = 2;   // 確認の画面で名前を並べる本数（3 本目からは「ほか N 本」）

export function ReceiveScreen() {
  const { translator } = useI18n();
  const { update } = useSettings();
  const routinesValue = useRoutines();
  const storage = useStorage();
  const [step, setStep] = useState<Step>('pick');
  const [busy, setBusy] = useState(false);
  const [alert, setAlert] = useState<PickAlert>(null);
  const [saveFailed, setSaveFailed] = useState(false);
  const [plan, setPlan] = useState<ImportPlan | null>(null);
  const input = useRef<HTMLInputElement>(null);
  const latestRoutines = useRef(routinesValue);
  latestRoutines.current = routinesValue;
  const alive = useRef(true);
  const inLine = isLineInAppBrowser();

  // 選んだ・届いたファイルを読む（20MB を超えたら読まない）→ 形を確かめる → 受け取る計画（ID の付け直し・置き換え・上限）。
  const take = async (file: File) => {
    setBusy(true);
    setAlert(null);
    try {
      const parsed = parseBundle(await readBundleText(file));
      if (!alive.current) return;
      if (!parsed.ok) { setAlert(parsed.problem === 'too-new' ? { kind: 'too-new' } : { kind: 'broken', file: file.name }); return; }
      const packs = await listVoicePacks(storage);
      if (!alive.current) return;
      const next = planImport(parsed, packs, latestRoutines.current.routines, { now: new Date().toISOString(), newId: makeId });
      if (next.voiceBlocked) { setAlert({ kind: 'voice-limit' }); return; }
      setPlan(next);
      setSaveFailed(false);
      setStep('confirm');
    } catch (error) {
      if (alive.current) setAlert(error instanceof BundleTooLargeError ? { kind: 'too-large' } : { kind: 'broken', file: file.name });
    } finally {
      if (alive.current) setBusy(false);
    }
  };

  useEffect(() => {
    void takeInboxFile().then(file => { if (file && alive.current) void take(file); });
    return () => { alive.current = false; };
  }, []);

  // 「ファイルを選ぶ」: 隠した <input type=file> を同じタップの中で開く。同じファイルを続けて選べるように値は毎回空にする（Ruling 83-5）。
  const choose = () => { if (!busy) input.current?.click(); };
  const picked = (element: HTMLInputElement) => {
    const file = element.files?.[0];
    element.value = '';
    if (file) void take(file);
  };

  // 「使う」: 新しい物を書けてから古い物を消す。声はその場で選ぶ（S7 で選び直させない）。
  const use = () => {
    if (!plan || busy) return;
    setBusy(true);
    const writer = {
      saveVoice: (pack: VoicePackRecord) => saveVoicePack(storage, pack),
      deleteVoice: (id: string) => deleteVoicePack(storage, id),
      saveRoutine: latestRoutines.current.save,
      removeRoutine: latestRoutines.current.remove
    };
    void commitImport(plan, writer).then(ok => {
      if (!alive.current) return;
      setBusy(false);
      if (!ok) { setSaveFailed(true); return; }
      if (plan.voice) update(chooseVoice(packSelection(plan.voice.pack)));
      setStep('done');
    });
  };

  const header = (
    <header class="dc-focus-head"><BackButton screen="S10" /><span class="dc-focus-title">{translator.t('receiveTitle')}</span></header>
  );

  if (step === 'done' && plan) {
    const voice = plan.voice?.pack;
    return (
      <main class="dc-screen dc-screen--focus dc-screen--dock dc-receive" data-screen="S10" data-step="done">
        {header}
        <h1 class="dc-heading">{translator.t('receivedTitle')}</h1>
        <p class="dc-body">
          {voice ? voiceNext(translator, voice) : translator.t('receivedRoutines')}
        </p>
        <div class="dc-dock">
          <button type="button" class="dc-button dc-primary dc-dock-button"
            onClick={() => replaceRoute({ screen: voice ? 'S1' : 'S3', params: {} })}>
            <span>{translator.t(voice ? 'goHome' : 'seeRoutines')}</span><Icon name="chev-r" />
          </button>
        </div>
      </main>
    );
  }

  if (step === 'confirm' && plan) {
    const voice = plan.voice;
    const names = plan.routines.map(item => item.routine.name);
    const replacesRoutines = plan.routines.some(item => item.replaces);
    return (
      <main class="dc-screen dc-screen--focus dc-screen--dock dc-receive" data-screen="S10" data-step="confirm">
        {header}
        <h1 class="dc-heading">{translator.t('inThisFile')}</h1>
        {saveFailed && (
          <div class="dc-alert" role="status" data-alert="save-failed">
            <div class="dc-alert-text dc-alert-text--solo"><b>{translator.t('saveFailed')}</b></div>
          </div>
        )}
        <div class="dc-receive-list" data-body>
          {voice && <div class="dc-receive-item"><Icon name="voice" /><span>{voiceLine(translator, voice.pack, plan.distributor)}</span></div>}
          {names.length > 0 && <div class="dc-receive-item"><Icon name="list" /><span>{routinesLine(translator, names)}</span></div>}
          {plan.rejected.slice(0, NAMES_SHOWN).map((item, index) => (
            <div key={index} class="dc-receive-note">{translator.t('routineUnreadable', { name: item.name ?? translator.t('routineNoName') })}</div>
          ))}
          {plan.rejected.length > NAMES_SHOWN && <div class="dc-receive-note">{translator.t('routinesUnreadableMore', { count: plan.rejected.length - NAMES_SHOWN })}</div>}
          {voice?.replaces && <div class="dc-receive-note">{replaceLine(translator, voice.pack, plan.distributor)}</div>}
          {replacesRoutines && <div class="dc-receive-note">{translator.t('replaceRoutines')}</div>}
          {plan.routinesOverLimit > 0 && <div class="dc-receive-note">{translator.t('routinesOverLimit', { max: ROUTINES_MAX, count: plan.routinesOverLimit })}</div>}
        </div>
        <div class="dc-dock dc-receive-dock">
          <button type="button" class="dc-button dc-button--secondary" disabled={busy} onClick={() => { setPlan(null); setStep('pick'); }}>
            <span>{translator.t('cancel')}</span>
          </button>
          <button type="button" class="dc-button dc-primary" disabled={busy} onClick={use}>
            <Icon name="check" /><span>{translator.t('useFile')}</span>
          </button>
        </div>
      </main>
    );
  }

  const lineUrl = inLine ? externalBrowserUrl() : null;
  return (
    <main class="dc-screen dc-screen--focus dc-receive" data-screen="S10" data-step="pick">
      {header}
      {alert ? <PickNotice alert={alert} /> : inLine && (
        <div class="dc-alert" role="status" data-alert="in-line">
          <div class="dc-alert-text">{translator.t('openedInLine')}</div>
          {lineUrl && <a class="dc-alert-action" data-alert-action href={lineUrl}><span>{translator.t('openInSafari')}</span><Icon name="chev-r" /></a>}
        </div>
      )}
      <input ref={input} type="file" accept={BUNDLE_ACCEPT} hidden onChange={event => picked(event.currentTarget as HTMLInputElement)} />
      <button type="button" class="dc-button dc-primary dc-wide dc-receive-choose" disabled={busy} onClick={choose}>
        <span>{translator.t(busy ? 'readingFile' : 'chooseFile')}</span>
      </button>
      <div class="dc-receive-help" data-body>
        <div>{translator.t('receiveHelpLine')}</div>
        <div>{translator.t('receiveHelpRecents')}</div>
      </div>
    </main>
  );
}

function PickNotice({ alert }: { alert: NonNullable<PickAlert> }) {
  const { translator } = useI18n();
  const text = alert.kind === 'broken' ? translator.t('fileBroken', { file: alert.file })
    : alert.kind === 'too-large' ? translator.t('fileTooLarge')
    : alert.kind === 'too-new' ? translator.t('fileTooNew')
    : translator.t('voiceLimitReached', { count: VOICE_LIMITS.packs });
  return (
    <div class="dc-alert" role="status" data-alert={alert.kind}>
      <div class={`dc-alert-text${alert.kind === 'too-new' ? '' : ' dc-alert-text--solo'}`}><b>{text}</b></div>
      {alert.kind === 'too-new' && <button type="button" class="dc-alert-action" data-alert-action onClick={reloadPage}>{translator.t('updateApp')}</button>}
    </div>
  );
}

function languageName(translator: AppTranslator, pack: VoicePackRecord): string {
  return translator.t(pack.locale === 'ja-JP' ? 'languageJapanese' : 'languageEnglish');
}

// 「山田さんの声（日本語）· 12 語」。旧画面の単体パック（名前の無いファイル）は声の名前のまま。
function voiceLine(translator: AppTranslator, pack: VoicePackRecord, distributor: { name: string } | null): string {
  const values = { language: languageName(translator, pack), count: packProgress(pack).recorded };
  return distributor ? translator.t('fileVoiceFrom', { name: distributor.name, ...values }) : translator.t('fileVoiceNamed', { name: pack.name, ...values });
}

function replaceLine(translator: AppTranslator, pack: VoicePackRecord, distributor: { name: string } | null): string {
  return distributor ? translator.t('replaceVoiceFrom', { name: distributor.name }) : translator.t('replaceVoiceNamed', { name: pack.name });
}

function voiceNext(translator: AppTranslator, pack: VoicePackRecord): string {
  return pack.distributor ? translator.t('receivedVoiceFrom', { name: pack.distributor.name }) : translator.t('receivedVoiceNamed', { name: pack.name });
}

// 「ルーティン 2 本: ワルツのルーティン、タンゴ前のワルツ」。3 本目からは「ほか N 本」。
function routinesLine(translator: AppTranslator, names: string[]): string {
  const separator = translator.locale === 'ja-JP' ? '、' : ', ';
  const shown = names.slice(0, NAMES_SHOWN).join(separator);
  const list = names.length > NAMES_SHOWN ? translator.t('namesAndMore', { names: shown, count: names.length - NAMES_SHOWN }) : shown;
  return names.length === 1 ? translator.t('fileRoutinesOne', { names: list }) : translator.t('fileRoutines', { count: names.length, names: list });
}
